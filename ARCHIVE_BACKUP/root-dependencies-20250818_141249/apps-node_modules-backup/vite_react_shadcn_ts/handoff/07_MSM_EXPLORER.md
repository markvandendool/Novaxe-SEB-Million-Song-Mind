# MSM EXPLORER - HARMONIC ORACLE

## 🎵 Overview

The Million Song Mind (MSM) Explorer is Harmonic Oracle's primary data exploration interface, providing tabular analysis, filtering, and harmonic visualization of Data3 Pure datasets.

---

## 🏗️ Architecture

### Component Hierarchy
```
MillionSongMind.tsx (main page)
├── DataTable (sortable/filterable table)
├── ChordChart (harmonic profile visualization)
├── FilterControls (search, genre, key filters)
├── SearchInterface (chord progression search)
└── ExportControls (CSV/JSON export)
```

### Data Flow
```
CSV Upload → Format Detection → Parsing → Validation → MSM Display
     ↓              ↓              ↓          ↓           ↓
 Index.tsx → cpmlParser.ts → HUV Processing → State → UI Render
```

```typescript
const MillionSongMind: React.FC = () => {
  const [data, setData] = useState<HarmonicData[]>([]);
```

### HUV Data Aggregation
```typescript
// Aggregate HUV vectors across multiple songs
const aggregateHUVData = (songs: HarmonicData[]): AggregatedHUV => {
  const aggregated: Record<string, HUVVector> = {};
  
  // Initialize all slots
  HARMONIC_SLOTS.forEach(slot => {
    aggregated[slot] = { total: 0, root: 0, first: 0, second: 0, third: 0 };
  });
  
  // Sum across all songs
  songs.forEach(song => {
    Object.entries(song.huvBySlot).forEach(([slot, vector]) => {
      if (aggregated[slot]) {
        aggregated[slot].total += vector.total;
        aggregated[slot].root += vector.root;
        aggregated[slot].first += vector.first;
        aggregated[slot].second += vector.second;
        aggregated[slot].third += vector.third;
      }
    });
  });
  
  return {
    bySlot: aggregated,
    totalSongs: songs.length,
    totalChords: Object.values(aggregated).reduce((sum, vector) => sum + vector.total, 0),
    mostCommonChords: Object.entries(aggregated)
      .sort(([,a], [,b]) => b.total - a.total)
      .slice(0, 10)
      .map(([slot, vector]) => ({ slot, count: vector.total }))
  };
};
```

```typescript
// Advanced search with chord progression matching
const searchChordProgressions = (data: HarmonicData[], query: string): HarmonicData[] => {
  const queryChords = query.split(/\s+/).filter(Boolean);
  
  return data.filter(song => {
    const songChords = song.chords.split(/\s+/);
    
    // Exact sequence match
    if (containsSequence(songChords, queryChords)) {
      return true;
    }
    
    // Roman numeral match
    const romanChords = song.romanNumerals.split(/\s+/);
    if (containsSequence(romanChords, queryChords)) {
      return true;
    }
    
    // Fuzzy match (allows gaps)
    if (containsFuzzySequence(songChords, queryChords)) {
      return true;
    }
    
    return false;
  });
};

const containsSequence = (haystack: string[], needle: string[]): boolean => {
  for (let i = 0; i <= haystack.length - needle.length; i++) {
    if (needle.every((chord, j) => 
      normalizeChord(haystack[i + j]) === normalizeChord(chord)
    )) {
      return true;
    }
  }
  return false;
};
```

```typescript
    let direction: 'asc' | 'desc' = 'asc';

      direction = 'desc';
    }
    
    onSort({ key, direction });

    if (!sortConfig) return data;
    
    return [...data].sort((a, b) => {
      const aValue = a[sortConfig.key];
      const bValue = b[sortConfig.key];
      
      if (aValue < bValue) return sortConfig.direction === 'asc' ? -1 : 1;
      if (aValue > bValue) return sortConfig.direction === 'asc' ? 1 : -1;
      return 0;
```

```typescript
// src/components/HarmonicProfileMini.tsx
const HarmonicProfileMini: React.FC<{
  data: Record<string, HUVVector>;
  maxHeight?: number;
}> = ({ data, maxHeight = 40 }) => {
  
  const chartData = HARMONIC_SLOTS.map(slot => ({
    slot,
    value: data[slot]?.total || 0
  }));
  
  const maxValue = Math.max(...chartData.map(d => d.value));
  
  return (
    <div className="harmonic-profile-mini flex items-end space-x-1" style={{ height: maxHeight }}>
      {chartData.map(({ slot, value }) => {
        const height = maxValue > 0 ? (value / maxValue) * maxHeight : 0;
        const color = getChordColor(slot);
        
        return (
          <div
            key={slot}
            className="flex-1 min-w-0 rounded-t-sm"
            style={{
              height: `${height}px`,
              backgroundColor: color,
              opacity: value > 0 ? 1 : 0.2
            }}
            title={`${slot}: ${value}`}
          />
        );
      })}
    </div>
  );
};

const getChordColor = (slot: string): string => {
  // Major chords
  if (['I', 'IV', 'V', 'I7', 'V(7)'].includes(slot)) return '#3b82f6'; // Blue
  
  // Minor chords
  if (['ii', 'iii', 'vi', 'i', 'iv', 'v'].includes(slot)) return '#10b981'; // Green
  
  // Diminished/Half-diminished
  if (slot.includes('º') || slot.includes('ø')) return '#f59e0b'; // Yellow
  
  // Borrowed/Chromatic
  if (slot.includes('#') || slot.includes('b')) return '#ef4444'; // Red
  
  // Other
  if (slot === 'Other') return '#6b7280'; // Gray
  
  return '#8b5cf6'; // Purple (default)
};
```

```typescript
// src/components/VirtualDataTable.tsx (future implementation)
const VirtualDataTable: React.FC<{
  data: HarmonicData[];
  rowHeight: number;
  containerHeight: number;
}> = ({ data, rowHeight, containerHeight }) => {
  
  const [scrollTop, setScrollTop] = useState(0);
  
  const visibleStart = Math.floor(scrollTop / rowHeight);
  const visibleEnd = Math.min(
    visibleStart + Math.ceil(containerHeight / rowHeight) + 1,
    data.length
  );
  
  const visibleItems = data.slice(visibleStart, visibleEnd);
  const totalHeight = data.length * rowHeight;
  const offsetY = visibleStart * rowHeight;
  
  return (
    <div 
      className="virtual-table-container"
      style={{ height: containerHeight, overflowY: 'auto' }}
      onScroll={(e) => setScrollTop(e.currentTarget.scrollTop)}
    >
      <div style={{ height: totalHeight, position: 'relative' }}>
        <div style={{ transform: `translateY(${offsetY}px)` }}>
          {visibleItems.map((item, index) => (
            <DataRow 
              key={visibleStart + index}
              data={item}
              style={{ height: rowHeight }}
            />
          ))}
        </div>
      </div>
    </div>
  );
};
```

```typescript
export const exportToCSV = (data: HarmonicData[], filename: string = 'harmonic_data.csv') => {
  const headers = [
    'Song', 'Artist', 'Key', 'Chords', 'Roman Numerals', 
    'Genre', 'Decade', 'Chosen By', 'Other Rate', 'KS Score',

        return `"${vector.total},${vector.root},${vector.first},${vector.second},${vector.third}"`;

  const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = filename;
  link.click();
  URL.revokeObjectURL(link.href);
};
```
