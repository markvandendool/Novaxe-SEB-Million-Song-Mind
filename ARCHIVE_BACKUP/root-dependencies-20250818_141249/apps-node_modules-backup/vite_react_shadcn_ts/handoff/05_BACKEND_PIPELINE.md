def parse_cpml_chords(cpml_string: str) -> List[str]:
    """Extract chords from CPML section tags."""
    import re
    
    # Remove section tags: <verse_1>Am F C G</verse_1> → Am F C G
    pattern = r'<[^>]+>(.*?)</[^>]+>'
    matches = re.findall(pattern, cpml_string)
    
    chord_text = ' '.join(matches)
    chords = chord_text.split()
    
    return [normalize_chord(chord) for chord in chords if chord]

def normalize_chord(chord: str) -> str:
    """Standardize chord notation."""
    # Am → Am, F → FM, C → CM
    chord = chord.strip()
    if chord and not chord.endswith('m') and not any(c in chord for c in ['7', 'sus', 'add', 'dim', 'aug']):
        if chord[-1].isupper():
            chord += 'M'  # C → CM
    return chord
```

def detect_key_final_boss(chords: List[str]) -> Tuple[str, float, float]:
    """
    Select key that minimizes 'Other' chord classifications.
    
    Returns:
        (chosen_key, other_rate, ks_score)
    """
    candidates = ALL_24_KEYS  # ['C', 'G', 'D', 'A', 'E', 'B', 'F#', 'C#', 'F', 'Bb', 'Eb', 'Ab', 'Am', 'Em', 'Bm', 'F#m', 'C#m', 'G#m', 'Ebm', 'Bbm', 'Fm', 'Cm', 'Gm', 'Dm']
    
    # Get Krumhansl-Schmuckler scores for all keys
    ks_scores = krumhansl_schmuckler_analysis(chords)
    
    best_key = None
    best_other_rate = float('inf')
    best_ks_score = -1
    
    for key in candidates:
        # Convert chords to Roman numerals in this key
        romans = convert_to_romans(chords, key)
        
        # Build 27-slot HUV histogram
        huv_histogram = build_huv_histogram(romans)
        
        # Calculate "Other" rate
        total_chords = sum(huv_histogram.values())
        other_count = huv_histogram.get('Other', 0)
        other_rate = other_count / total_chords if total_chords > 0 else 1.0
        
        # Select key with lowest other_rate, KS score as tiebreaker
        if (other_rate < best_other_rate or 
            (other_rate == best_other_rate and ks_scores[key] > best_ks_score)):
            best_key = key
            best_other_rate = other_rate
            best_ks_score = ks_scores[key]
    
    return best_key, best_other_rate, best_ks_score
```

**3. HUV Vector Generation**
```python
def generate_huv_vectors(romans: List[str], key: str) -> Dict[str, str]:
    """
    Generate 27-slot HUV vectors with inversion tracking.
    
    Returns:
        Dictionary mapping slot names to vector strings
    """
    # Initialize 27 slots
    slots = {
        # Major mode (12 slots)
        'I': [0, 0, 0, 0, 0], 'ii': [0, 0, 0, 0, 0], 'iii': [0, 0, 0, 0, 0],
        'IV': [0, 0, 0, 0, 0], 'V': [0, 0, 0, 0, 0], 'vi': [0, 0, 0, 0, 0],
        'viiº': [0, 0, 0, 0, 0], 'I7': [0, 0, 0, 0, 0], 'iiiø': [0, 0, 0, 0, 0],
        'II(7)': [0, 0, 0, 0, 0], '#ivø': [0, 0, 0, 0, 0], 'III(7)': [0, 0, 0, 0, 0],
        '#vº': [0, 0, 0, 0, 0], 'VI(7)': [0, 0, 0, 0, 0], '#iº': [0, 0, 0, 0, 0],
        'VII(7)': [0, 0, 0, 0, 0], '#iiº': [0, 0, 0, 0, 0], 'V(7)': [0, 0, 0, 0, 0],
        'viiº': [0, 0, 0, 0, 0],
        
        # Minor mode (7 slots)  
        'i': [0, 0, 0, 0, 0], 'iiø': [0, 0, 0, 0, 0], 'bIII': [0, 0, 0, 0, 0],
        'iv': [0, 0, 0, 0, 0], 'v': [0, 0, 0, 0, 0], 'bVI': [0, 0, 0, 0, 0],
        'bVII': [0, 0, 0, 0, 0],
        
        # Special slots (2 slots)
        'V(b9)': [0, 0, 0, 0, 0], 'Other': [0, 0, 0, 0, 0]
    }
    
    for roman in romans:
        slot_name = map_roman_to_slot(roman, key)
        inversion = detect_inversion(roman)
        
        # Update vector: [total, root, first, second, third]
        slots[slot_name][0] += 1  # total
        slots[slot_name][inversion + 1] += 1  # inversion position
    
    # Convert to string format
    return {
        slot: ','.join(map(str, vector))
        for slot, vector in slots.items()
    }
```

    Async enrichment with Spotify metadata.
    """
    import spotipy
    from spotipy.oauth2 import SpotifyClientCredentials

        client_id=SPOTIFY_CLIENT_ID,

    sp = spotipy.Spotify(client_credentials_manager=client_credentials_manager)
    
    # Process in batches to respect API limits
    batch_size = 50
    for i in range(0, len(df), batch_size):
        batch = df.iloc[i:i+batch_size]
        
        for idx, row in batch.iterrows():
            if pd.isna(row['artist_name']) or pd.isna(row['song_name']):
                try:
                    # Search Spotify
                    query = f"track:{row['song_name']} artist:{row['artist_name']}"
                    results = sp.search(q=query, type='track', limit=1)
                    
                    if results['tracks']['items']:
                        track = results['tracks']['items'][0]
                        df.at[idx, 'artist_name'] = track['artists'][0]['name']
                        df.at[idx, 'song_name'] = track['name']
                        df.at[idx, 'artist_url'] = track['artists'][0]['external_urls']['spotify']
                        df.at[idx, 'song_url'] = track['external_urls']['spotify']
                        
                except Exception as e:
                    logger.warning(f"Spotify lookup failed for row {idx}: {e}")
                    continue
        
        # Rate limiting
        await asyncio.sleep(0.1)
        
        # Save progress checkpoint
        if i % 500 == 0:

## 🔍 Data Validation & Quality

### Schema Validation
```python
def validate_data3_schema(df: pd.DataFrame) -> List[str]:
    """
    Validate Data3 Pure CSV schema compliance.
    
    Returns:
        List of validation errors
    """
    errors = []
    
    # Required columns
    required_cols = [
        'id', 'chords', 'key', 'roman_numerals', 'harmonic_fingerprint',
        'artist_name', 'song_name', 'chosen_by', 'other_rate', 'ks_score'
    ]
    
    # 27 HUV slots
    huv_slots = [
        'I', 'ii', 'iii', 'IV', 'V', 'vi', 'viiº', 'I7', 'iiiø', 'II(7)',
        '#ivø', 'III(7)', '#vº', 'VI(7)', '#iº', 'VII(7)', '#iiº', 'V(7)', 'viiº',
        'i', 'iiø', 'bIII', 'iv', 'v', 'bVI', 'bVII', 'V(b9)', 'Other'
    ]
    
    all_required = required_cols + huv_slots
    
    # Check column presence
    missing_cols = [col for col in all_required if col not in df.columns]
    if missing_cols:
        errors.append(f"Missing columns: {missing_cols}")
    
    # Validate HUV vectors
    for slot in huv_slots:
        if slot in df.columns:
            invalid_vectors = []
            for idx, cell in df[slot].items():
                if not validate_huv_vector(cell):
                    invalid_vectors.append(idx)
            
            if invalid_vectors:
                errors.append(f"Invalid HUV vectors in {slot}: rows {invalid_vectors[:5]}...")
    
    return errors

def validate_huv_vector(cell: str) -> bool:
    """Validate individual HUV vector format."""
    if pd.isna(cell) or cell == '0,0,0,0,0':
        return True
    
    try:
        parts = str(cell).split('|')
        for part in parts:
            numbers = [int(n.strip()) for n in part.split(',')]
            if len(numbers) != 5:
                return False
            if any(n < 0 for n in numbers):
                return False
        return True
    except:
        return False
```

### Spotify Web API
```python
class SpotifyEnricher:
    def __init__(self, client_id: str, client_secret: str):
        self.client_credentials_manager = SpotifyClientCredentials(
            client_id=client_id,
            client_secret=client_secret
        )
        self.sp = spotipy.Spotify(
            client_credentials_manager=self.client_credentials_manager
        )
    
    async def search_track(self, artist: str, song: str) -> Optional[Dict]:
        """Search for track metadata."""
        try:
            query = f"track:\"{song}\" artist:\"{artist}\""
            results = self.sp.search(q=query, type='track', limit=1)
            
            if results['tracks']['items']:
                track = results['tracks']['items'][0]
                return {
                    'artist_name': track['artists'][0]['name'],
                    'song_name': track['name'],
                    'artist_url': track['artists'][0]['external_urls']['spotify'],
                    'song_url': track['external_urls']['spotify'],
                    'spotify_artist_id': track['artists'][0]['id'],
                    'spotify_song_id': track['id']
                }
            
            return None
            
        except Exception as e:
            logger.warning(f"Spotify search failed: {e}")
            return None
```

### Async Processing
```python
async def process_batch_async(batch: List[Dict], enricher: SpotifyEnricher) -> List[Dict]:
    """Process batch of records asynchronously."""
    tasks = []
    
    for record in batch:
        if not record.get('artist_name') or not record.get('song_name'):
            task = enricher.search_track(
                record.get('artist_id', ''),
                record.get('song_id', '')
            )
            tasks.append(task)
        else:
            tasks.append(asyncio.create_task(asyncio.sleep(0)))  # No-op
    
    results = await asyncio.gather(*tasks, return_exceptions=True)
    
    # Merge results back
    for i, result in enumerate(results):
        if isinstance(result, dict):
            batch[i].update(result)
    
    return batch
```

### Progress Tracking
```python
class ProgressTracker:
    def __init__(self, total: int, description: str = "Processing"):
        self.total = total
        self.current = 0
        self.description = description
        self.start_time = time.time()
    
    def update(self, increment: int = 1):
        """Update progress and log status."""
        self.current += increment
        
        if self.current % 100 == 0 or self.current == self.total:
            elapsed = time.time() - self.start_time
            rate = self.current / elapsed if elapsed > 0 else 0
            eta = (self.total - self.current) / rate if rate > 0 else 0
            
            logger.info(
                f"{self.description}: {self.current}/{self.total} "
                f"({self.current/self.total*100:.1f}%) "
                f"Rate: {rate:.1f}/sec ETA: {eta:.0f}s"
            )
```
