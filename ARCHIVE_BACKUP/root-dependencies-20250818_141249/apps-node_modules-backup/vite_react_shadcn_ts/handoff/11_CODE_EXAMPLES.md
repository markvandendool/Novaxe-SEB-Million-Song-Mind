```typescript
// src/utils/cpmlParser.ts
export function parseUnifiedCSVData(csvContent: string): ParsedData {
  // CRITICAL: BOM stripping for UTF-8 files
  const cleanContent = csvContent.replace(/^\uFEFF/, '');
  
  const lines = cleanContent.split('\n').filter(line => line.trim());
  if (lines.length < 2) {
    throw new Error('CSV must contain header and data rows');
  }
  
  const headers = lines[0].split(',').map(h => h.trim().replace(/"/g, ''));
  const format = detectCSVFormat(headers);
  
  console.log(`parseUnifiedCSVData:start - Format: ${format}, Rows: ${lines.length - 1}`);
  
  switch (format) {
    case 'data3':
      return parseData3CSV(cleanContent);
    case 'datanaught':
      return parseDatanaughtCSV(cleanContent);
    default:
      throw new Error(`Unsupported CSV format: ${format}`);
  }
}
```

### HUV Vector Processing
```typescript
// src/utils/cpmlParser.ts
function parseHUVVector(cell: string): HUVVector {
  if (!cell || cell === '0,0,0,0,0') {
    return { total: 0, root: 0, first: 0, second: 0, third: 0 };
  }
  
  try {
    // Split on pipe for extended vectors
    const parts = cell.split('|');
    const primary = parts[0].split(',').map(n => parseInt(n.trim()));
    
    if (primary.length !== 5) {
      console.warn(`Invalid HUV vector format: ${cell}`);
      return { total: 0, root: 0, first: 0, second: 0, third: 0 };
    }
    
    const vector: HUVVector = {
      total: primary[0],
      root: primary[1],
      first: primary[2], 
      second: primary[3],
      third: primary[4]
    };
    
    // Parse extended color-tone vectors if present
    if (parts.length > 1) {
      vector.extensions = parts.slice(1).map(part => {
        const nums = part.split(',').map(n => parseInt(n.trim()));
        return nums.length === 5 ? nums : null;
      }).filter(Boolean);
    }
    
    return vector;
  } catch (error) {
    console.error(`Failed to parse HUV vector: ${cell}`, error);
    return { total: 0, root: 0, first: 0, second: 0, third: 0 };
  }
}
```

```typescript
// src/utils/cpmlParser.ts
function parseData3Row(row: string[], headers: string[]): HarmonicData | null {
  try {
    const data: any = {};
    
    // Map basic columns
    headers.forEach((header, index) => {
      const value = row[index]?.trim().replace(/"/g, '') || '';
      data[header.toLowerCase()] = value;
    });
    
    // Build HUV vectors for each slot
    const huvBySlot: Record<string, HUVVector> = {};
    HARMONIC_SLOTS.forEach(slot => {
      const cellValue = data[slot.toLowerCase()] || '0,0,0,0,0';
      huvBySlot[slot] = parseHUVVector(cellValue);
    });
    
    // Convert to HarmonicData format
    const harmonicData: HarmonicData = {
      id: data.id || '',
      chords: data.chords || '',
      artist: data.artist_name || data.artist || 'Unknown Artist',
      song: data.song_name || data.song || 'Unknown Song',
      key: data.key || 'C',
      romanNumerals: data.roman_numerals || '',
      harmonicFingerprint: data.harmonic_fingerprint || '',
      decade: data.decade || '',
      genre: data.genres || data.genre || '',
      
      // Audit columns (Final Boss)
      chosenBy: data.chosen_by || 'unknown',
      otherRate: parseFloat(data.other_rate || '0'),
      ksScore: parseFloat(data.ks_score || '0'),
      
      // HUV data
      huvBySlot,
      
      // Computed properties
      inversionSubBars: calculateInversionSubBars(huvBySlot),
      totalChords: Object.values(huvBySlot).reduce((sum, vector) => sum + vector.total, 0)
    };
    
    console.log(`parseData3CSV:row - ID: ${harmonicData.id}, Chords: ${harmonicData.totalChords}`);
    return harmonicData;
    
  } catch (error) {
    console.error('Failed to parse Data3 row:', error);
    return null;
  }
}
```

### CSV Processing Error Recovery
```typescript
// src/utils/cpmlParser.ts
function parseCSVWithErrorRecovery(csvContent: string): ParseResult {
  const errors: string[] = [];
  const validRows: HarmonicData[] = [];
  const invalidRows: number[] = [];
  
  try {
    const lines = csvContent.split('\n');
    const headers = lines[0].split(',').map(h => h.trim());
    
    for (let i = 1; i < lines.length; i++) {
      const line = lines[i].trim();
      if (!line) continue;
      
      try {
        const row = parseCSVLine(line);
        const harmonicData = parseData3Row(row, headers);
        
        if (harmonicData) {
          validRows.push(harmonicData);
        } else {
          invalidRows.push(i);
          errors.push(`Row ${i}: Failed to parse harmonic data`);
        }
        
      } catch (rowError) {
        invalidRows.push(i);
        errors.push(`Row ${i}: ${rowError.message}`);
        console.warn(`Skipping invalid row ${i}:`, rowError);
      }
    }
    
    return {
      data: validRows,
      errors,
      invalidRows,
      totalRows: lines.length - 1,
      validRows: validRows.length
    };
    
  } catch (error) {
    throw new Error(`CSV parsing failed: ${error.message}`);
  }
}
```

```typescript
// src/utils/validation.ts
export function validateHUVVector(vector: string): ValidationResult {
  if (!vector || vector.trim() === '') {
    return { valid: true, warnings: ['Empty HUV vector'] };
  }
  
  if (vector === '0,0,0,0,0') {
    return { valid: true, warnings: [] }; // Empty slot is valid
  }
  
  try {
    const parts = vector.split('|');
    const errors: string[] = [];
    const warnings: string[] = [];
    
    for (let i = 0; i < parts.length; i++) {
      const part = parts[i].trim();
      const numbers = part.split(',').map(n => {
        const num = parseInt(n.trim());
        if (isNaN(num)) {
          errors.push(`Invalid number: "${n}" in part ${i + 1}`);
          return 0;
        }
        return num;
      });
      
      if (numbers.length !== 5) {
        errors.push(`Part ${i + 1} must have exactly 5 numbers, got ${numbers.length}`);
        continue;
      }
      
      // Validate constraints
      const [total, root, first, second, third] = numbers;
      
      if (total < 0) {
        errors.push(`Total cannot be negative: ${total}`);
      }
      
      if (root + first + second + third > total) {
        warnings.push(`Inversion sum (${root + first + second + third}) exceeds total (${total})`);
      }
      
      if (total > 0 && root + first + second + third === 0) {
        warnings.push(`Total is ${total} but no inversions specified`);
      }
    }
    
    return {
      valid: errors.length === 0,
      errors,
      warnings
    };
    
  } catch (error) {
    return {
      valid: false,
      errors: [`HUV vector parsing failed: ${error.message}`],
      warnings: []
    };
  }
}

interface ValidationResult {
  valid: boolean;
  errors?: string[];
  warnings?: string[];
}
```

### Data3 Schema Validation
```typescript
// src/utils/validation.ts
export function validateData3Schema(data: any[]): SchemaValidationResult {
  if (!Array.isArray(data) || data.length === 0) {
    return {
      valid: false,
      errors: ['Data must be a non-empty array'],
      summary: { totalRows: 0, validRows: 0, errorRows: 0 }
    };
  }
  
  const requiredColumns = [
    'id', 'chords', 'key', 'roman_numerals', 'harmonic_fingerprint',
    'artist_name', 'song_name'
  ];
  
  const huvSlots = [
    'I', 'ii', 'iii', 'IV', 'V', 'vi', 'viiº', 'I7', 'iiiø', 'II(7)',
    '#ivø', 'III(7)', '#vº', 'VI(7)', '#iº', 'VII(7)', '#iiº', 'V(7)', 'viiº',
    'i', 'iiø', 'bIII', 'iv', 'v', 'bVI', 'bVII', 'V(b9)', 'Other'
  ];
  
  const errors: string[] = [];
  const warnings: string[] = [];
  let validRows = 0;
  let errorRows = 0;
  
  // Check first row for schema compliance
  if (data.length > 0) {
    const firstRow = data[0];
    const missingRequired = requiredColumns.filter(col => !(col in firstRow));
    const missingHUV = huvSlots.filter(slot => !(slot in firstRow));
    
    if (missingRequired.length > 0) {
      errors.push(`Missing required columns: ${missingRequired.join(', ')}`);
    }
    
    if (missingHUV.length > 0) {
      errors.push(`Missing HUV slots: ${missingHUV.join(', ')}`);
    }
  }
  
  // Validate each row
  data.forEach((row, index) => {
    let rowValid = true;
    
    // Check required fields
    requiredColumns.forEach(col => {
      if (!row[col] || row[col].toString().trim() === '') {
        warnings.push(`Row ${index + 1}: Missing ${col}`);
      }
    });
    
    // Validate HUV vectors
    huvSlots.forEach(slot => {
      if (row[slot]) {
        const validation = validateHUVVector(row[slot]);
        if (!validation.valid) {
          errors.push(`Row ${index + 1}, ${slot}: ${validation.errors?.join(', ')}`);
          rowValid = false;
        }
        if (validation.warnings?.length > 0) {
          warnings.push(`Row ${index + 1}, ${slot}: ${validation.warnings.join(', ')}`);
        }
      }
    });
    
    if (rowValid) {
      validRows++;
    } else {
      errorRows++;
    }
  });
  
  return {
    valid: errors.length === 0,
    errors,
    warnings,
    summary: {
      totalRows: data.length,
      validRows,
      errorRows,
      warningRows: warnings.length
    }
  };
}

interface SchemaValidationResult {
  valid: boolean;
  errors: string[];
  warnings: string[];
  summary: {
    totalRows: number;
    validRows: number;
    errorRows: number;
    warningRows?: number;
  };
}
```
