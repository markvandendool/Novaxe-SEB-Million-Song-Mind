### Performance Test Data Generator
```typescript
// Generate large datasets for performance testing
export function generateLargeTestDataset(size: number): HarmonicData[] {
  const artists = ['Artist A', 'Artist B', 'Artist C', 'Artist D', 'Artist E'];
  const keys = ['C', 'G', 'D', 'A', 'E', 'B', 'F#', 'F', 'Bb', 'Eb', 'Ab', 'Db'];
  const progressions = [
    'I V vi IV', 'vi IV I V', 'I vi IV V', 'ii V I', 'I IV V I'
  ];
  
  return Array.from({ length: size }, (_, i) => ({
    id: `test_${i}`,
    artist: artists[i % artists.length],
    song: `Test Song ${i}`,
    key: keys[i % keys.length],
    chords: progressions[i % progressions.length],
    romanNumerals: progressions[i % progressions.length],
    harmonicFingerprint: `test_fingerprint_${i}`,
    decade: '2020s',
    genre: 'test',
    chosenBy: 'final_boss',
    otherRate: Math.random() * 0.3,
    ksScore: 0.5 + Math.random() * 0.4,
    huvBySlot: generateRandomHUVSlots(),
    inversionSubBars: [],
    totalChords: 4
  }));
}

function generateRandomHUVSlots(): Record<string, HUVVector> {
  const slots: Record<string, HUVVector> = {};
  
  HARMONIC_SLOTS.forEach(slot => {
    const total = Math.floor(Math.random() * 10);
    const root = Math.floor(Math.random() * total);
    const first = Math.floor(Math.random() * (total - root));
    const remaining = total - root - first;
    const second = Math.floor(Math.random() * remaining);
    const third = remaining - second;
    
    slots[slot] = { total, root, first, second, third };
  });
  
  return slots;
}
```

```typescript
export function measurePerformance(fn: () => void, label: string) {
```

### Data Validation Scripts
```typescript
// validate-test-data.ts
import { validateData3Schema } from '../src/utils/validation';
import { parseUnifiedCSVData } from '../src/utils/cpmlParser';

export async function validateAllTestData() {
  const testFiles = [
    'test/mock_data3_pure.csv',
    'Lovable Correspondences/Lovable.csv',
    'test/edge_cases.csv'
  ];
  
  for (const file of testFiles) {
    try {
      const content = await fs.readFile(file, 'utf-8');
      const parsed = parseUnifiedCSVData(content);
      const validation = validateData3Schema(parsed.data);
      
      console.log(`${file}:`, {
        valid: validation.valid,
        rows: validation.summary.totalRows,
        errors: validation.errors.length,
        warnings: validation.warnings.length
      });
      
      if (!validation.valid) {
        console.error(`❌ ${file} validation failed:`, validation.errors);
      } else {
        console.log(`✅ ${file} validation passed`);
      }
      
    } catch (error) {
      console.error(`❌ Failed to validate ${file}:`, error.message);
    }
  }
}
```
