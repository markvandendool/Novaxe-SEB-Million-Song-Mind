## 🔧 Data Processing Pipeline

### CSV Parsing (`src/utils/cpmlParser.ts`)
```typescript
// SENTINEL: parseUnifiedCSVData:start
export function parseUnifiedCSVData(csvContent: string): ParsedData {
  // BOM stripping
  const cleanContent = csvContent.replace(/^\uFEFF/, '');
  
  // Header processing
  const lines = cleanContent.split('\n');
  const headers = lines[0].split(',').map(h => h.trim());
  
  // Data3 vs Datanaught detection
  const format = detectCSVFormat(headers);
  
  if (format === 'data3') {
    return parseData3CSV(cleanContent);
  }
  
  return parseDatanaughtCSV(cleanContent);
}
```

### HUV Vector Processing
```typescript
function parseHUVVector(cell: string): HUVVector {
  if (!cell || cell === '0,0,0,0,0') {
    return { total: 0, root: 0, first: 0, second: 0, third: 0 };
  }
  
  const parts = cell.split('|');
  const primary = parts[0].split(',').map(n => parseInt(n.trim()));
  
  return {
    total: primary[0],
    root: primary[1], 
    first: primary[2],
    second: primary[3],
    third: primary[4],
    extensions: parts.slice(1) // Color-tone vectors
  };
}
```

### Format Detection (`src/lib/utils.ts`)
```typescript
// SENTINEL: harmonic_fingerprint
export function detectCSVFormat(headers: string[]): CSVFormat {
  const lowerHeaders = headers.map(h => h.toLowerCase());
  
  if (lowerHeaders.includes('harmonic_fingerprint')) {
    return 'data3';
  }
  
  if (lowerHeaders.includes('track') && lowerHeaders.includes('artist')) {
    return 'datanaught';
  }
  
  return 'unknown';
}
```

## 🔍 Error Handling

### Error Boundaries
```typescript
// src/components/ErrorBoundary.tsx
class ErrorBoundary extends React.Component {
  constructor(props) {
    super(props);
    this.state = { hasError: false };
  }
  
  static getDerivedStateFromError(error) {
    return { hasError: true };
  }
  
  componentDidCatch(error, errorInfo) {
    console.error('Error caught by boundary:', error, errorInfo);
  }
  
  render() {
    if (this.state.hasError) {
      return <ErrorFallback />;
    }
    
    return this.props.children;
  }
}
```

### Async Error Handling
```typescript
const handleAsyncOperation = async () => {
  try {
    setIsLoading(true);
    const result = await processData(data);
    setResult(result);
  } catch (error) {
    setError(error.message);
    console.error('Processing failed:', error);
  } finally {
    setIsLoading(false);
  }
};
```
