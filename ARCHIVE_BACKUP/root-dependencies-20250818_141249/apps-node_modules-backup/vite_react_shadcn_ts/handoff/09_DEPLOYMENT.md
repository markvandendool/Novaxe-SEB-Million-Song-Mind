## 🔄 CI/CD Pipeline

### GitHub Actions Workflow
```yaml
# .github/workflows/deploy.yml
name: Deploy to GitHub Pages

on:
  push:
    branches: [ main ]
  pull_request:
    branches: [ main ]

permissions:
  contents: read
  pages: write
  id-token: write

concurrency:
  group: "pages"
  cancel-in-progress: true

jobs:
  # Build job
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          
      - name: Setup pnpm
        uses: pnpm/action-setup@v2
        with:
          version: 8
          
      - name: Install dependencies
        run: pnpm install --frozen-lockfile
        
      - name: Type check
        run: pnpm type-check
        
      - name: Lint
        run: pnpm lint
        
      - name: Build
        run: pnpm build
        env:
          VITE_BUILD_MODE: production
          
      - name: Upload build artifacts
        uses: actions/upload-pages-artifact@v2
        with:
          path: ./dist

  # Deploy job (only on main branch)
  deploy:
    if: github.ref == 'refs/heads/main'
    needs: build
    runs-on: ubuntu-latest
    environment:
      name: github-pages
      url: ${{ steps.deployment.outputs.page_url }}
    steps:
      - name: Deploy to GitHub Pages
        id: deployment
        uses: actions/deploy-pages@v3
```

### PR Guard Workflow
```yaml
# .github/workflows/pr-guard.yml
name: PR Guard

on:
  pull_request:
    branches: [ main ]

jobs:
  guard:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout
        uses: actions/checkout@v4
        with:
          fetch-depth: 0
          
      - name: Setup Node.js
        uses: actions/setup-node@v4
        with:
          node-version: '20'
          
      - name: Setup pnpm
        uses: pnpm/action-setup@v2
        with:
          version: 8
          
      - name: Install dependencies
        run: pnpm install --frozen-lockfile
        
      - name: Build verification
        run: pnpm build
        
      - name: Validate sentinels
        run: |
          echo "Checking for required sentinel strings..."
          
          # MSM ingestion sentinels
          grep -q "msm:pendingUpload" src/pages/Index.tsx || (echo "❌ Missing sentinel: msm:pendingUpload" && exit 1)
          grep -q "parseUnifiedCSVData:start" src/utils/cpmlParser.ts || (echo "❌ Missing sentinel: parseUnifiedCSVData:start" && exit 1)
          grep -q "harmonic_fingerprint" src/lib/utils.ts || (echo "❌ Missing sentinel: harmonic_fingerprint" && exit 1)
          grep -q "V(7)" src/constants/harmony.ts || (echo "❌ Missing sentinel: V(7)" && exit 1)
          
          echo "✅ All sentinels present"
          
      - name: Check protected paths
        run: |
          # Get list of changed files
          CHANGED_FILES=$(git diff --name-only ${{ github.event.pull_request.base.sha }}..HEAD)
          
          # Check for braid modifications
          if echo "$CHANGED_FILES" | grep -E '^src/(components/braid|pages/Braid)'; then
            # Check for allow-braid label
            LABELS=$(gh pr view ${{ github.event.number }} --json labels --jq '.labels[].name')
            if ! echo "$LABELS" | grep -q "allow-braid"; then
              echo "❌ Braid files modified without 'allow-braid' label"
              echo "Changed braid files:"
              echo "$CHANGED_FILES" | grep -E '^src/(components/braid|pages/Braid)'
              exit 1
            fi
          fi
          
          # Check for backend modifications
          if echo "$CHANGED_FILES" | grep -E '^(src/utils/cpmlParser.ts|src/lib/utils.ts|scripts/|tools/)'; then
            LABELS=$(gh pr view ${{ github.event.number }} --json labels --jq '.labels[].name')
            if ! echo "$LABELS" | grep -q "allow-backend"; then
              echo "❌ Backend files modified without 'allow-backend' label"
              echo "Changed backend files:"
              echo "$CHANGED_FILES" | grep -E '^(src/utils/cpmlParser.ts|src/lib/utils.ts|scripts/|tools/)'
              exit 1
            fi
          fi
          
          echo "✅ Protected path checks passed"
        env:
          GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}
```

### Static Asset Handling
```typescript
// vite.config.ts - Asset optimization
export default defineConfig({
  build: {
    rollupOptions: {
      output: {
        assetFileNames: (assetInfo) => {
          const info = assetInfo.name.split('.');
          const ext = info[info.length - 1];
          
          if (/png|jpe?g|svg|gif|tiff|bmp|ico/i.test(ext)) {
            return `assets/images/[name]-[hash][extname]`;
          }
          
          if (/woff2?|eot|ttf|otf/i.test(ext)) {
            return `assets/fonts/[name]-[hash][extname]`;
          }
          
          return `assets/[name]-[hash][extname]`;
        }
      }
    }
  },
  
  // Asset processing
  assetsInclude: ['**/*.woff', '**/*.woff2', '**/*.ttf'],
  
  // Public directory handling
  publicDir: 'public'
});
```

### Image Optimization
```typescript
// src/components/ImageOptimizer.tsx
interface OptimizedImageProps {
  src: string;
  alt: string;
  width?: number;
  height?: number;
  loading?: 'lazy' | 'eager';
}

const OptimizedImage: React.FC<OptimizedImageProps> = ({
  src,
  alt,
  width,
  height,
  loading = 'lazy'
}) => {
  // Generate responsive image sources
  const srcSet = useMemo(() => {
    if (!width) return undefined;
    
    const sizes = [1, 1.5, 2]; // 1x, 1.5x, 2x pixel densities
    return sizes
      .map(density => `${src}?w=${width * density}&q=80 ${density}x`)
      .join(', ');
  }, [src, width]);

  return (
    <img
      src={`${src}?w=${width}&q=80`}
      srcSet={srcSet}
      alt={alt}
      width={width}
      height={height}
      loading={loading}
      decoding="async"
    />
  );
};
```

### Preloading Strategy
```typescript
// src/utils/preloader.ts
export class AssetPreloader {
  private static preloadedAssets = new Set<string>();
  
  static preloadRoute(routePath: string) {
    // Preload route-specific assets
    const routeAssets = this.getRouteAssets(routePath);
    
    routeAssets.forEach(asset => {
      if (!this.preloadedAssets.has(asset)) {
        this.preloadAsset(asset);
        this.preloadedAssets.add(asset);
      }
    });
  }
  
  private static preloadAsset(src: string) {
    const link = document.createElement('link');
    link.rel = 'prefetch';
    link.href = src;
    document.head.appendChild(link);
  }
  
  private static getRouteAssets(routePath: string): string[] {
    const assetMap: Record<string, string[]> = {
      '/braid-tonal': [
        '/assets/braid_tonalities.json',
        '/assets/fonts/ShareTechMono-Regular.ttf'
      ],
      '/million-song-mind': [
        '/assets/chart-icons.svg'
      ]
    };
    
    return assetMap[routePath] || [];
  }
}
```

### Error Tracking Setup
```typescript
// src/utils/errorTracking.ts
interface ErrorReport {
  message: string;
  stack?: string;
  url: string;
  line?: number;
  column?: number;
  timestamp: string;
  userAgent: string;
  userId?: string;
}

class ErrorTracker {
  private static endpoint = '/api/errors';
  
  static init() {
    // Global error handler
    window.addEventListener('error', (event) => {
      this.reportError({
        message: event.message,
        stack: event.error?.stack,
        url: event.filename || window.location.href,
        line: event.lineno,
        column: event.colno,
        timestamp: new Date().toISOString(),
        userAgent: navigator.userAgent
      });
    });
    
    // Promise rejection handler
    window.addEventListener('unhandledrejection', (event) => {
      this.reportError({
        message: `Unhandled Promise Rejection: ${event.reason}`,
        stack: event.reason?.stack,
        url: window.location.href,
        timestamp: new Date().toISOString(),
        userAgent: navigator.userAgent
      });
    });
  }
  
  private static async reportError(error: ErrorReport) {
    try {
      await fetch(this.endpoint, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(error)
      });
    } catch (e) {
      console.error('Failed to report error:', e);
    }
  }
}

// Initialize in main.tsx
ErrorTracker.init();
```

### Analytics Integration
```typescript
// src/utils/analytics.ts
interface AnalyticsEvent {
  action: string;
  category: string;
  label?: string;
  value?: number;
}

class Analytics {
  private static initialized = false;
  
  static init() {
    if (this.initialized || !ENV.analyticsId) return;
    
    // Google Analytics 4
    const script = document.createElement('script');
    script.async = true;
    script.src = `https://www.googletagmanager.com/gtag/js?id=${ENV.analyticsId}`;
    document.head.appendChild(script);
    
    window.gtag = window.gtag || function() {
      (window.gtag.q = window.gtag.q || []).push(arguments);
    };
    
    window.gtag('js', new Date());
    window.gtag('config', ENV.analyticsId);
    
    this.initialized = true;
  }
  
  static trackEvent({ action, category, label, value }: AnalyticsEvent) {
    if (!this.initialized) return;
    
    window.gtag('event', action, {
      event_category: category,
      event_label: label,
      value: value
    });
  }
  
  static trackPageView(path: string) {
    if (!this.initialized) return;
    
    window.gtag('config', ENV.analyticsId, {
      page_path: path
    });
  }
}

// Usage in components
Analytics.trackEvent({
  action: 'file_upload',
  category: 'data_processing',
  label: 'data3_csv',
  value: dataSize
});
```

### Health Check Endpoint
```typescript
// src/utils/healthCheck.ts
interface HealthStatus {
  status: 'healthy' | 'degraded' | 'unhealthy';
  timestamp: string;
  checks: {
    build: boolean;
    assets: boolean;
    storage: boolean;
    performance: boolean;
  };
  metrics: {
    loadTime: number;
    memoryUsage: number;
    errorRate: number;
  };
}

export async function performHealthCheck(): Promise<HealthStatus> {
  const startTime = performance.now();
  
  const checks = {
    build: await checkBuildIntegrity(),
    assets: await checkAssetAvailability(),
    storage: checkStorageAvailability(),
    performance: await checkPerformanceMetrics()
  };
  
  const allHealthy = Object.values(checks).every(Boolean);
  const loadTime = performance.now() - startTime;
  
  return {
    status: allHealthy ? 'healthy' : 'degraded',
    timestamp: new Date().toISOString(),
    checks,
    metrics: {
      loadTime,
      memoryUsage: (performance as any).memory?.usedJSHeapSize || 0,
      errorRate: getErrorRate()
    }
  };
}

async function checkBuildIntegrity(): Promise<boolean> {
  try {
    // Check if critical assets are available
    const response = await fetch('/assets/braid_tonalities.json');
    return response.ok;
  } catch {
    return false;
  }
}
```

### Performance Monitoring
```typescript
// src/utils/performanceMonitor.ts
class PerformanceMonitor {
  private static metrics: Array<{
    name: string;
    value: number;
    timestamp: number;
  }> = [];
  
  static measurePageLoad() {
    if ('PerformanceObserver' in window) {
      const observer = new PerformanceObserver((list) => {
        list.getEntries().forEach((entry) => {
          this.recordMetric(entry.name, entry.duration);
        });
      });
      
      observer.observe({ entryTypes: ['navigation', 'measure'] });
    }
  }
  
  static measureComponent(name: string, fn: () => void) {
    const start = performance.now();
    fn();
    const duration = performance.now() - start;
    
    this.recordMetric(`component_${name}`, duration);
  }
  
  private static recordMetric(name: string, value: number) {
    this.metrics.push({
      name,
      value,
      timestamp: Date.now()
    });
    
    // Keep only last 100 metrics
    if (this.metrics.length > 100) {
      this.metrics = this.metrics.slice(-100);
    }
  }
  
  static getMetrics() {
    return [...this.metrics];
  }
}
```
