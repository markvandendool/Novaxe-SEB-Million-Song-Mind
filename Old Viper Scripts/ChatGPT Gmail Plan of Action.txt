#!/usr/bin/env python3
"""
Spotify Metadata Enrichment Pass (v1.0)
======================================
- Runs separately after theory analysis
- Loads previous data3 CSV with numeric IDs
- Queries Spotify metadata for artist/track names
- Resumes from where it left off using enrichment checkpoint
- Supports VPN switching by pausing on rate limit

Usage: python enrich_spotify.py
"""

import os
import time
import json
import pandas as pd
import aiohttp
import asyncio
from pathlib import Path
from tqdm import tqdm
from your_original_modules import (
    ProductionLogger,
    IndustrialSpotifyClient,
    save_checkpoint,
    load_checkpoint,
)

ENRICHMENT_CSV = "data3_final.csv"
OUTPUT_CSV = "data3_enriched.csv"
CHECKPOINT_FILE = "spotify_enrichment_checkpoint.json"
RATE_LIMIT_FLAG = "rate_limit_triggered.flag"

logger = ProductionLogger("spotify_enricher")

# --------------------------------------
# Async Spotify metadata enrichment
# --------------------------------------
async def enrich_spotify_metadata(df):
    client = IndustrialSpotifyClient()
    client.load_token()

    # Track progress
    checkpoint = load_checkpoint(CHECKPOINT_FILE)
    enriched = checkpoint.get("enriched_ids", {})
    pending_rows = df[~df["spotify_song_id"].astype(str).isin(enriched.keys())]
    logger.info(f"Need to enrich {len(pending_rows)} Spotify records")

    enriched_data = []
    try:
        for i, row in tqdm(pending_rows.iterrows(), total=len(pending_rows), desc="Spotify Metadata"):
            track_id = row["spotify_song_id"]
            try:
                track, artist = await client.get_track_and_artist(track_id)
                enriched_row = row.copy()
                enriched_row["track_name"] = track
                enriched_row["artist_name"] = artist
                enriched_data.append(enriched_row)

                enriched[str(track_id)] = True

                # Save every 100 rows
                if len(enriched_data) % 100 == 0:
                    temp_df = pd.DataFrame(enriched_data)
                    merged = pd.concat([df[~df["spotify_song_id"].isin(temp_df["spotify_song_id"])], temp_df], ignore_index=True)
                    merged.to_csv(OUTPUT_CSV, index=False)
                    save_checkpoint(CHECKPOINT_FILE, enriched, len(enriched))

            except client.RateLimitException as e:
                logger.warning(f"Rate limited. Wait suggested: {e.retry_after}s")
                Path(RATE_LIMIT_FLAG).touch()
                return
            except Exception as e:
                logger.error(f"Failed to enrich {track_id}: {e}")

    finally:
        if enriched_data:
            final_df = pd.DataFrame(enriched_data)
            merged = pd.concat([df[~df["spotify_song_id"].isin(final_df["spotify_song_id"])], final_df], ignore_index=True)
            merged.to_csv(OUTPUT_CSV, index=False)
            save_checkpoint(CHECKPOINT_FILE, enriched, len(enriched))
            logger.info(f"🔄 Saved enriched CSV with {len(enriched)} Spotify records")

# --------------------------------------
# Entrypoint
# --------------------------------------
if __name__ == "__main__":
    if not os.path.exists(ENRICHMENT_CSV):
        logger.error(f"Missing input: {ENRICHMENT_CSV}")
        exit(1)

    df = pd.read_csv(ENRICHMENT_CSV)

    if os.path.exists(RATE_LIMIT_FLAG):
        logger.warning("⏳ Rate limit previously hit. Waiting for manual VPN switch...")
        os.system("afplay /System/Library/Sounds/Glass.aiff")
        input("🔁 Switch VPN and press [Y] to continue: ")
        os.remove(RATE_LIMIT_FLAG)

    asyncio.run(enrich_spotify_metadata(df))
    logger.info("✅ Spotify enrichment complete.")
