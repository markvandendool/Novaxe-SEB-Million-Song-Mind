@dataclass
class MachineSpecs:
    """Machine specifications and work assignment"""
    model_name: str
    model_identifier: str
    physical_cores: int
    logical_cores: int
    memory_gb: float
    hostname: str
    work_range: Tuple[int, Optional[int]]  # (start_row, end_row) - None means to end
    output_suffix: str
    cpu_target_percent: float = 90.0

class MachineDetector:
    """Automatic machine detection and work assignment"""
    
    @staticmethod
    def detect_machine() -> MachineSpecs:
        """Detect current machine and assign work range"""
        
        try:
            # Get hardware info
            if platform.system() == 'Darwin':  # macOS
                # Get model identifier
                result = subprocess.run(['system_profiler', 'SPHardwareDataType'], 
                                      capture_output=True, text=True)
                hardware_info = result.stdout
                
                model_name = ""
                model_identifier = ""
                memory_gb = 0
                
                for line in hardware_info.split('\n'):
                    if 'Model Name:' in line:
                        model_name = line.split(':')[1].strip()
                    elif 'Model Identifier:' in line:
                        model_identifier = line.split(':')[1].strip()
                    elif 'Memory:' in line:
                        memory_str = line.split(':')[1].strip()
                        memory_gb = float(memory_str.split()[0])
                
                # Get CPU info
                physical_cores = psutil.cpu_count(logical=False)
                logical_cores = psutil.cpu_count(logical=True)
                hostname = platform.node()
                
                # Machine-specific work assignment
                if model_identifier == 'MacPro7,1':
                    # Mac Pro 2019 - Handle first 300k rows
                    return MachineSpecs(
                        model_name=model_name,
                        model_identifier=model_identifier,
                        physical_cores=physical_cores,
                        logical_cores=logical_cores,
                        memory_gb=memory_gb,
                        hostname=hostname,
                        work_range=(0, 299999),
                        output_suffix="macpro",
                        cpu_target_percent=90.0
                    )
                elif 'Mac14' in model_identifier or 'MacStudio' in model_identifier:
                    # Mac Studio M2 Max - Handle remaining rows
                    return MachineSpecs(
                        model_name=model_name,
                        model_identifier=model_identifier,
                        physical_cores=physical_cores,
                        logical_cores=logical_cores,
                        memory_gb=memory_gb,
                        hostname=hostname,
                        work_range=(300000, None),
                        output_suffix="studio",
                        cpu_target_percent=90.0
                    )
                else:
                    # Unknown Mac - process everything (fallback)
                    return MachineSpecs(
                        model_name=model_name,
                        model_identifier=model_identifier,
                        physical_cores=physical_cores,
                        logical_cores=logical_cores,
                        memory_gb=memory_gb,
                        hostname=hostname,
                        work_range=(0, None),
                        output_suffix="unknown",
                        cpu_target_percent=75.0  # Be more conservative
                    )
            else:
                # Non-macOS fallback
                return MachineSpecs(
                    model_name="Unknown",
                    model_identifier="Unknown",
                    physical_cores=psutil.cpu_count(logical=False),
                    logical_cores=psutil.cpu_count(logical=True),
                    memory_gb=psutil.virtual_memory().total / (1024**3),
                    hostname=platform.node(),
                    work_range=(0, None),
                    output_suffix="generic",
                    cpu_target_percent=75.0
                )
        
        except Exception as e:
            print(f"Machine detection failed: {e}")
            # Emergency fallback
            return MachineSpecs(
                model_name="Fallback",

                hostname="unknown",

                output_suffix="fallback",
                cpu_target_percent=50.0
            )

class PerformanceLogger:
    """Ultra-fast logging optimized for high CPU usage"""
    
    def __init__(self, machine_specs: MachineSpecs):
        self.machine = machine_specs
        self.start_time = time.time()
        self.metrics = defaultdict(int)
        
        # Setup logging
        log_filename = f"viper_dual_{machine_specs.output_suffix}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.log"
        
        logging.basicConfig(
            level=logging.INFO,
            format='%(asctime)s | %(levelname)s | %(message)s',
            handlers=[
                logging.FileHandler(log_filename, encoding='utf-8'),
                logging.StreamHandler(sys.stdout)
            ]
        )
        self.logger = logging.getLogger(f'VIPER_{machine_specs.output_suffix.upper()}')
        
        self.info(f"🔥 VIPER DUAL-MACHINE INITIALIZED ON {machine_specs.hostname}")
        self.info(f"💻 Machine: {machine_specs.model_name} ({machine_specs.model_identifier})")
        self.info(f"⚡ CPU: {machine_specs.physical_cores}C/{machine_specs.logical_cores}T")
        self.info(f"💾 RAM: {machine_specs.memory_gb:.0f}GB")
        self.info(f"🎯 Work Range: {machine_specs.work_range[0]:,} to {machine_specs.work_range[1] or 'END'}")
        self.info(f"📁 Output: data3_{machine_specs.output_suffix}.csv")
    
    def info(self, msg: str): 
        self.logger.info(f"🎵 {msg}")
        self.metrics['log_messages'] += 1
    
    def success(self, msg: str): 
        self.logger.info(f"✅ {msg}")
    
    def warning(self, msg: str): 
        self.logger.warning(f"⚠️  {msg}")
    
    def error(self, msg: str): 
        self.logger.error(f"❌ {msg}")
    
    def critical(self, msg: str): 
        self.logger.critical(f"💥 {msg}")
    
    def metric(self, key: str, value: int = 1):
        """Track metrics thread-safely"""
        self.metrics[key] += value
    
    def performance_report(self) -> Dict[str, Any]:
        """Generate performance report"""
        elapsed = time.time() - self.start_time
        return {
            'runtime_minutes': elapsed / 60,
            'songs_processed': self.metrics.get('songs_processed', 0),
            'songs_per_second': self.metrics.get('songs_processed', 0) / max(elapsed, 1),
            'errors': self.metrics.get('errors', 0),
            'total_metrics': dict(self.metrics)
        }

    def __init__(self):
        # High-performance caches
        self._chord_cache = {}

    def normalize_note(self, note: str) -> str:
        """Normalize note names"""
        if not note:
            return note
        return self.ENHARMONIC_MAP.get(note.strip(), note.strip())
    
    @lru_cache(maxsize=50000)
    def parse_chord_fast(self, chord_str: str) -> Optional[Dict[str, Any]]:
        """Ultra-fast chord parsing optimized for speed"""
        
        if not chord_str or pd.isna(chord_str) or not isinstance(chord_str, str):
            return None
        
        chord_str = chord_str.strip()
        if not chord_str:
            return None
        
        # Fast regex pattern
        pattern = (
            r'^([A-G][b#♯♭]?)'                    # Root note
            r'(maj|min|dim|aug|m|°|\+|M|△)?'       # Basic quality
            r'(\d+)?'                             # Extension number
            r'([b#♯♭]\d+|add\d+|sus[24]?|alt|no3d?|\+|°)*'  # Alterations
            r'(/([A-G][b#♯♭]?))?$'                # Bass note
        )
        
        match = re.match(pattern, chord_str, re.IGNORECASE)
        if not match:
            return None
        
        root, quality, extension, alterations, _, bass = match.groups()
        
        # Normalize components
        root = self.normalize_note(root)
        bass = self.normalize_note(bass) if bass else None
        
        # Fast chord type determination
        chord_type = self._determine_chord_type_fast(quality, extension, alterations)
        
        # Get intervals
        chord_data = self.CHORD_DATABASE.get(chord_type, self.CHORD_DATABASE['maj'])
        intervals = chord_data['intervals']
        
        return {
            'root': root,
            'chord_type': chord_type,
            'bass': bass,
            'intervals': intervals,
            'original': chord_str,
            'quality_family': chord_data['quality'],
            'complexity': chord_data['complexity']
        }
    
    def _determine_chord_type_fast(self, quality: str, extension: str, alterations: str) -> str:
        """Fast chord type determination"""
        
        quality = quality.lower() if quality else ''
        extension = extension if extension else ''
        alterations = alterations.lower() if alterations else ''
        
        # Handle power chords first
        if 'no3' in alterations:
            return 'no3'
        
        # Base quality
        if quality in ['min', 'm']:
            base = 'min'
        elif quality in ['dim', '°']:
            base = 'dim'
        elif quality in ['aug', '+']:
            base = 'aug'
        else:
            base = 'maj'
        
        # Extensions
        if extension == '7':
            if base == 'maj':
                chord_type = 'dom7' if 'maj' not in quality else 'maj7'
            else:
                chord_type = f'{base}7'
        elif extension in ['9', '11', '13']:
            chord_type = f'{base}{extension}' if base != 'maj' else f'dom{extension}'
        else:
            chord_type = base
        
        # Alterations
        if 'sus4' in alterations:
            chord_type = '7sus4' if '7' in chord_type else 'sus4'
        elif 'sus2' in alterations:
            chord_type = '7sus2' if '7' in chord_type else 'sus2'
        elif 'add9' in alterations:
            chord_type = 'add9' if base == 'maj' else 'madd9'
        elif 'add6' in alterations:
            chord_type = 'add6' if base == 'maj' else 'min6'
        elif 'alt' in alterations:
            chord_type = 'dom7alt'
        elif 'b9' in alterations:
            chord_type = chord_type.replace('dom7', 'dom7b9')
        elif '#9' in alterations:
            chord_type = chord_type.replace('dom7', 'dom7#9')
        
        # Validate
        if chord_type not in self.CHORD_DATABASE:
            chord_type = base if base in self.CHORD_DATABASE else 'maj'
        
        return chord_type
    
    def detect_key_fast(self, chord_sequence: List[str]) -> Tuple[str, bool, float]:
        """Ultra-fast key detection"""
        
        if not chord_sequence:
            return 'C', True, 0.0
        
        # Cache check
        sequence_key = '|'.join(chord_sequence[:100])
        if sequence_key in self._key_cache:
            return self._key_cache[sequence_key]
        
        # Initialize pitch class histogram
        pitch_classes = np.zeros(12, dtype=np.float64)
        valid_chords = 0
        
        for chord_str in chord_sequence:
            if chord_str.startswith('<'):
                continue
            
            parsed = self.parse_chord_fast(chord_str)
            if not parsed:
                continue
            
            valid_chords += 1
            root_semitone = self.NOTE_TO_SEMITONE.get(parsed['root'], 0)
            
            # Weight root heavily
            pitch_classes[root_semitone] += 3.0
            
            # Add chord tones
            for interval in parsed['intervals']:
                if interval == 0:
                    continue
                semitone = (root_semitone + interval) % 12
                pitch_classes[semitone] += 1.0
        
        if valid_chords == 0:
            result = ('C', True, 0.0)
            self._key_cache[sequence_key] = result
            return result
        
        # Normalize
        pitch_classes = pitch_classes / np.sum(pitch_classes)
        
        # Fast correlation calculation
        major_correlations = np.array([
            np.corrcoef(pitch_classes, profile)[0, 1] if np.var(pitch_classes) > 0 else 0
            for profile in self._major_profiles
        ])
        minor_correlations = np.array([
            np.corrcoef(pitch_classes, profile)[0, 1] if np.var(pitch_classes) > 0 else 0
            for profile in self._minor_profiles
        ])
        
        # Handle NaN
        major_correlations = np.nan_to_num(major_correlations, nan=0.0)
        minor_correlations = np.nan_to_num(minor_correlations, nan=0.0)
        
        # Find best key
        best_major_idx = np.argmax(major_correlations)
        best_minor_idx = np.argmax(minor_correlations)
        
        if major_correlations[best_major_idx] >= minor_correlations[best_minor_idx]:
            result = (self.CHROMATIC_NOTES[best_major_idx], True, float(major_correlations[best_major_idx]))
        else:
            result = (self.CHROMATIC_NOTES[best_minor_idx], False, float(minor_correlations[best_minor_idx]))
        
        self._key_cache[sequence_key] = result
        return result
    
    def generate_roman_numerals_fast(self, chord_sequence: List[str], key: str, is_major: bool) -> List[str]:
        """Ultra-fast Roman numeral generation"""
        
        romans = []
        key_semitone = self.NOTE_TO_SEMITONE.get(key, 0)
        scale_degrees = self.MAJOR_SCALE_DEGREES if is_major else self.MINOR_SCALE_DEGREES
        
        for chord_str in chord_sequence:
            if chord_str.startswith('<'):
                romans.append(chord_str)
                continue
            
            parsed = self.parse_chord_fast(chord_str)
            if not parsed:
                romans.append('?')
                continue
            
            root_semitone = self.NOTE_TO_SEMITONE.get(parsed['root'], 0)
            degree = (root_semitone - key_semitone) % 12
            
            # Get base roman
            if degree in scale_degrees:
                roman = scale_degrees[degree]
                
                # Quick quality adjustment
                quality_family = parsed.get('quality_family', 'major')
                if quality_family == 'minor' and roman.isupper():
                    roman = roman.lower()
                elif quality_family == 'major' and roman.islower():
                    roman = roman.upper()
                
                # Add extensions
                chord_type = parsed['chord_type']
                if '7' in chord_type:
                    roman += '7'
                elif '9' in chord_type:
                    roman += '9'
                
                romans.append(roman)
            else:
                # Chromatic chord
                if parsed.get('quality_family') == 'dominant':
                    target_degree = (degree + 5) % 12
                    if target_degree in scale_degrees:
                        romans.append(f"V7/{scale_degrees[target_degree]}")
                        continue
                
                # Fallback
                if degree < 6:
                    romans.append(f"b{degree + 1}")
                else:
                    romans.append(f"#{degree - 5}")
        
        return romans
    
    def generate_huv_fingerprint_fast(self, chord_sequence: List[str]) -> str:
        """Ultra-fast HUV fingerprint generation"""
        
        huv_vectors = []
        
        for chord_str in chord_sequence:
            if chord_str.startswith('<'):
                huv_vectors.append(chord_str)
                continue
            
            parsed = self.parse_chord_fast(chord_str)
            if not parsed:
                huv_vectors.append('0.083,0.083,0.083,0.083,0.083,0.083,0.083,0.083,0.083,0.083,0.083,0.083')
                continue
            
            # Generate 12-dimensional HUV vector
            vector = np.zeros(12)
            root_semitone = self.NOTE_TO_SEMITONE.get(parsed['root'], 0)
            
            # Root gets maximum weight
            vector[root_semitone] = 1.0
            
            # Chord tones with decreasing weights
            for i, interval in enumerate(parsed['intervals']):
                if interval == 0:
                    continue
                semitone = (root_semitone + interval) % 12
                weight = max(0.3, 0.8 - (i * 0.1))
                vector[semitone] += weight
            
            # Normalize
            if np.sum(vector) > 0:
                vector = vector / np.sum(vector)
            
            huv_vectors.append(','.join(f"{x:.3f}" for x in vector))
        
        return '|'.join(huv_vectors)
    
    def extract_cpml_sequence_fast(self, cpml_string: str) -> List[str]:
        """Ultra-fast CPML sequence extraction"""
        
        if not cpml_string or pd.isna(cpml_string):
            return []
        
        # Fast regex pattern
        pattern = r'(<[^>]*>|[A-G][b#♯♭]?(?:maj|min|dim|aug|m|°|\+|sus|add|no3d?|M|△|\d+)*(?:[b#♯♭]\d+|alt|sus[24]?)*(?:/[A-G][b#♯♭]?)?)'
        
        matches = re.findall(pattern, cpml_string, re.IGNORECASE)
        
        # Quick cleaning
        cleaned = []
        for match in matches:
            match = match.strip()
            if match:
                if match.startswith('<') and not match.endswith('>'):
                    match += '>'
                cleaned.append(match)
        
        return cleaned

    def _calculate_optimal_workers(self) -> int:
        """Calculate optimal number of workers for target CPU usage"""
        
        # For CPU-intensive work, use slightly more workers than logical cores
        # to account for I/O wait time and maximize CPU utilization
        base_workers = self.machine.logical_cores
        
        if self.machine.cpu_target_percent >= 90:
            # Aggressive: Use 1.2x logical cores for maximum throughput
            workers = int(base_workers * 1.2)

            workers = int(base_workers * 1.25)
        else:
            # Conservative: Use 90% of logical cores

    async def process_data2_to_data3_lightning(self, input_file: str, output_file: str) -> Dict[str, Any]:
        """Lightning-fast data2 to data3 conversion with maximum CPU utilization"""
        
        start_time = time.time()
        
        # Banner
        self.logger.info("🚀" + "="*80)
        self.logger.success("⚡ LIGHTNING-FAST DATA3 CREATION - MAXIMUM PERFORMANCE MODE")

        # Load and filter data based on machine assignment
        self.logger.info("📁 Loading and filtering input data...")
        df = pd.read_csv(input_file, encoding='utf-8', dtype={'id': 'Int64'})
        
        total_rows_original = len(df)
        start_row, end_row = self.machine.work_range
        
        if end_row is not None:
            df = df.iloc[start_row:end_row+1]
        else:
            df = df.iloc[start_row:]
        
        total_rows_assigned = len(df)
        
        self.logger.success(f"📊 Data loaded: {total_rows_assigned:,} songs assigned to this machine (of {total_rows_original:,} total)")
        
        # Define exact data3 column structure
        data3_columns = [
            'id', 'chords', 'release_date', 'genres', 'decade', 'rock_genre',

            'artist_name', 'artist_url', 'song_name', 'song_url',
            'key', 'roman_numerals', 'harmonic_fingerprint'

        songs_data = df.to_dict('records')

        batch_size = max(10, len(songs_data) // (self.num_workers * 4))  # Ensure good work distribution
        song_batches = [songs_data[i:i+batch_size] for i in range(0, len(songs_data), batch_size)]
        
        self.logger.info(f"🔥 Processing {len(song_batches)} batches with {self.num_workers} workers")
        self.logger.info(f"⚡ Estimated processing time: {len(songs_data) / (50 * self.num_workers):.1f} minutes")
        
        # Process batches in parallel with maximum CPU utilization
        processed_songs = []
        batch_start = time.time()
        
        with ProcessPoolExecutor(max_workers=self.num_workers) as executor:
            # Submit all batches
            future_to_batch = {
                executor.submit(process_song_batch_wrapper, batch, self.music_theory): i 
                for i, batch in enumerate(song_batches)
            }
            
            # Collect results as they complete
            completed_batches = 0

                except Exception as e:
                    self.logger.error(f"Batch {batch_idx} failed: {e}")

                    failed_batch = song_batches[batch_idx]
                    for song in failed_batch:
                        song.update({

                            'roman_numerals': 'error',
                            'harmonic_fingerprint': '',
                            'artist_name': 'PENDING',

                            'song_name': 'PENDING',
                            'song_url': f"https://open.spotify.com/track/{song.get('spotify_song_id', '')}"
                        })
                    processed_songs.extend(failed_batch)

        result_df = pd.DataFrame(processed_songs)
        
        # Ensure exact column order
        result_df = result_df.reindex(columns=data3_columns, fill_value='')
        
        # Save results
        self.logger.info(f"💾 Saving results to {output_file}...")
        result_df.to_csv(output_file, index=False, encoding='utf-8', na_rep='')

        final_count = len(result_df)
        songs_per_second = final_count / total_time
        
        # Quality analysis
        successful_keys = len(result_df[~result_df['key'].isin(['No Harmony Data', 'Parse Error', 'Analysis Error', 'Processing Error'])])
        success_rate = successful_keys / final_count if final_count > 0 else 0
        
        results = {
            'machine_info': {
                'hostname': self.machine.hostname,
                'model': self.machine.model_name,
                'cores': f"{self.machine.physical_cores}C/{self.machine.logical_cores}T",

                'work_range': self.machine.work_range
            },
            'processing_stats': {
                'input_file': input_file,
                'output_file': output_file,
                'total_songs_processed': final_count,
                'processing_time_minutes': total_time / 60,
                'songs_per_second': songs_per_second,
                'success_rate': success_rate,
                'workers_used': self.num_workers
            },
            'quality_metrics': {
                'successful_key_detection': successful_keys,
                'key_detection_rate': success_rate,
                'total_errors': self.logger.metrics.get('errors', 0)
            }
        }
        
        # Victory report
        self.logger.info("🚀" + "="*80)
        self.logger.success("🎉 LIGHTNING-FAST PROCESSING COMPLETED!")
        self.logger.info("🚀" + "="*80)
        self.logger.success(f"📊 Processed: {final_count:,} songs in {total_time/60:.1f} minutes")
        self.logger.success(f"⚡ Speed: {songs_per_second:.1f} songs/second")
        self.logger.success(f"🎯 Success Rate: {success_rate:.1%}")
        self.logger.success(f"💻 Machine: {self.machine.hostname} ({self.machine.output_suffix})")
        self.logger.success(f"📁 Output: {output_file}")
        self.logger.info("🎵 Ready for Spotify metadata enrichment on 2012 iMac!")
        self.logger.info("🚀" + "="*80)
        
        return results
