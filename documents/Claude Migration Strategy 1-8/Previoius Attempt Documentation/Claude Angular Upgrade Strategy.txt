export class PianoModernComponent {
  private midiService = inject(MidiService);
  
  // Signals for reactive state
  keys = signal(this.generateKeys());
  activeNotes = signal(new Set<string>());
  
  // Computed for derived state
  isPlaying = computed(() => this.activeNotes().size > 0);
  
  constructor() {
    // Effect for side effects
    effect(() => {
      if (this.isPlaying()) {
        console.log('Piano active with', this.activeNotes().size, 'notes');
      }
    });
  }
  
  async playNote(note: string) {
    await this.midiService.playNote(note);
    this.activeNotes.update(notes => new Set(notes).add(note));
  }
  
  stopNote(note: string) {
    this.midiService.stopNote(note);
    this.activeNotes.update(notes => {
      const updated = new Set(notes);
      updated.delete(note);
      return updated;
    });
  }

Web MIDI API Implementation (Modern, No Dependencies)
typescript// services/web-midi.service.ts
@Injectable({ providedIn: 'root' })
export class WebMidiService {
  private midiAccess = signal<MIDIAccess | null>(null);
  private audioContext: AudioContext | null = null;
  
  async initialize(): Promise<void> {
    // Feature detection
    if (!navigator.requestMIDIAccess) {
      console.warn('Web MIDI API not supported, falling back to Web Audio');
      return this.initializeWebAudioFallback();
    }
    
    try {
      // Request MIDI access with sysex for full compatibility
      const access = await navigator.requestMIDIAccess({ sysex: true });
      this.midiAccess.set(access);
      
      // Setup audio context with autoplay handling
      this.audioContext = new (window.AudioContext || (window as any).webkitAudioContext)();
      
      // Handle autoplay policy
      document.addEventListener('click', () => {
        if (this.audioContext?.state === 'suspended') {
          this.audioContext.resume();
        }
      }, { once: true });
      
    } catch (error) {
      console.error('MIDI access denied:', error);
      return this.initializeWebAudioFallback();
    }
  }
  
  private initializeWebAudioFallback(): void {
    // Pure Web Audio fallback for browsers without MIDI
    this.audioContext = new (window.AudioContext || (window as any).webkitAudioContext)();
    console.log('Using Web Audio API fallback');
  }
  
  playNote(note: string, velocity = 127): void {
    if (this.midiAccess()) {
      // Use real MIDI
      const outputs = this.midiAccess()!.outputs;
      outputs.forEach(output => {
        output.send([0x90, this.noteToMidi(note), velocity]);
      });
    } else if (this.audioContext) {
      // Web Audio fallback
      const oscillator = this.audioContext.createOscillator();
      oscillator.frequency.value = this.noteToFrequency(note);
      oscillator.connect(this.audioContext.destination);
      oscillator.start();
      oscillator.stop(this.audioContext.currentTime + 0.5);
    }
  }
}

    autoFix: () => execSync('npm ci --force')
  },
  {
    pattern: /Property 'subscribe' does not exist/,
    solution: 'Convert to signals: toSignal(observable$)',
    autoFix: () => runCodemod('rxjs-to-signals')
  },
  {
    pattern: /MIDI\.js.*not found/,
    solution: 'Replace with WebMidiService',
    autoFix: () => replaceMidiImports()
  },
  {
    pattern: /Zone\.js.*already loaded/,
    solution: 'Remove duplicate zone.js import',
    autoFix: () => removeZoneDuplicates()
  },
  {
    pattern: /ngcc.*failed/,
    solution: 'Clear ngcc cache and rebuild',
    autoFix: () => execSync('rm -rf .angular && npx ngcc')
  }
];

// Run diagnostic
async function diagnoseAndFix(): Promise<void> {
  const buildOutput = execSync('ng build 2>&1', { encoding: 'utf8' });
  
  for (const error of ERROR_MATRIX) {
    if (error.pattern.test(buildOutput)) {
      console.log(`🔧 Detected: ${error.pattern}`);
      console.log(`💊 Solution: ${error.solution}`);
      
      if (error.autoFix) {
        console.log('🤖 Applying automatic fix...');
        error.autoFix();
      }
    }
  }
}
