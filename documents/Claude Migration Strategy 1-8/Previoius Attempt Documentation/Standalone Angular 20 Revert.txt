# Function to convert component back to module-based
convert_component_to_module() {
    local ts_file="$1"
    echo "  Converting: $ts_file"
    
    # Remove standalone and imports from @Component
    sed -i.bak '/standalone: true,/d; /imports: \[.*\],/d; /schemas: \[.*\],/d' "$ts_file"
    
    # Remove module imports that are now handled by modules
    sed -i.bak '/import { FormsModule }/d; /import { CommonModule }/d' "$ts_file"
}

# Convert all exercise components
for ts_file in $(find src/app/components/exercises -name "*.component.ts"); do
    convert_component_to_module "$ts_file"
done

# Convert shared components
for ts_file in $(find src/app/components/{metro,fifth-circle,midi-selector} -name "*.component.ts"); do
    convert_component_to_module "$ts_file"
done
