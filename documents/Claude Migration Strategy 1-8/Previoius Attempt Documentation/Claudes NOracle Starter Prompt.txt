```typescript
// src/app/features/msm/services/msm-bridge.service.ts
@Injectable({
  providedIn: 'root'
})
export class MsmBridgeService implements OnDestroy {
  private iframe$ = new BehaviorSubject<HTMLIFrameElement | null>(null);
  private ready$ = new BehaviorSubject<boolean>(false);
  private messageQueue: MsmMessage[] = [];
  private responseHandlers = new Map<string, (data: any) => void>();
  private subscriptions = new Subscription();

  constructor(
    private logger: Logger,
    private config: MsmConfig
  ) {}

  initialize(iframe: HTMLIFrameElement): Observable<boolean> {
    this.iframe$.next(iframe);
    
    // Setup message listener
    const messageHandler = (event: MessageEvent) => {
      if (event.origin !== new URL(this.config.url).origin) return;
      this.handleMessage(event.data);
    };
    
    window.addEventListener('message', messageHandler);
    this.subscriptions.add(() => window.removeEventListener('message', messageHandler));
    
    // Send handshake
    return this.sendHandshake().pipe(
      timeout(this.config.timeoutMs),
      retry(this.config.retryAttempts),
      tap(ready => this.ready$.next(ready)),
      catchError(error => {
        this.logger.error('MSM initialization failed', error);
        return of(false);
      })
    );
  }

  sendToMsm<T>(message: MsmMessage): Observable<T> {
    if (!this.ready$.value) {
      return throwError(() => new Error('MSM not ready'));
    }
    
    return new Observable(observer => {
      const messageId = this.generateMessageId();
      const messageWithId = { ...message, id: messageId };
      
      this.responseHandlers.set(messageId, (data: T) => {
        observer.next(data);
        observer.complete();
        this.responseHandlers.delete(messageId);
      });
      
      this.postMessage(messageWithId);
      
      // Cleanup timeout
      setTimeout(() => {
        if (this.responseHandlers.has(messageId)) {
          this.responseHandlers.delete(messageId);
          observer.error(new Error('MSM response timeout'));
        }
      }, this.config.timeoutMs);
    });
  }

  // State synchronization with debouncing
  syncState(stateType: string, data: any): void {
    const debouncedSync = this.getDebouncedSync(stateType);
    debouncedSync(data);
  }

  private getDebouncedSync(stateType: string): (data: any) => void {
    const debounceTime = this.getDebounceTime(stateType);
    return debounce((data: any) => {
      this.sendToMsm({
        type: 'state:update',
        payload: { stateType, data }
      }).subscribe();
    }, debounceTime);
  }

  private getDebounceTime(stateType: string): number {
    switch (stateType) {
      case 'notes': return 16; // High frequency note events
      case 'transport': return 60; // Transport position updates
      case 'analysis': return 200; // Heavy computation results
      default: return 100;
    }
  }
}
```
