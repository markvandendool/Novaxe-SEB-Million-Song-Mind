README.md

MillionSongMind — Angular 15
High-performance harmonic analysis and braid visualizations. Angular 15 + NgRx + Tailwind + Three.js.

Badges: CI, License, Commitizen-friendly.

Overview

Purpose: Explore tonal centers, progressions, and braid/torus topologies using deterministic state and optimized rendering.
Key Features:
Classic/Tonal/New braid visualizations (SVG + 3D).
Global key context with score/user lock.
Live metrics from corpus (presence, standalone, progressions).
SEO-first pages with canonical/meta/JSON-LD.
Reusable UI primitives themed via Tailwind semantic tokens (HSL).
Tech Stack

Angular 15, RxJS, NgRx, Tailwind, Three.js, Karma/Jasmine, Cypress, GitHub Actions.
Getting Started

Prereqs: Node 18 LTS, npm, Angular CLI 15.
Install: npm ci
Dev: npm start
Lint: npm run lint
Test: npm test
Build: npm run build
E2E (if added): npm run e2e
Architecture

Core services (SEO, data parsing, geometry).
Shared UI (headless components).
Feature modules (braid-classic, braid-tonal, braid-new2, live-metrics, braid-3d).
NgRx store slices (globalKey + router-store).
SEO

Title < 60 chars, meta description < 160 chars.
Single H1 per page. Canonical links and JSON-LD via services.
Lazy images, responsive layout.
Design System

Tailwind with semantic HSL tokens in styles.scss.
No hard-coded colors; use CSS vars.
State

NgRx with memoized selectors, OnPush, async pipes.
3D

Three.js integrated directly; careful cleanup in ngOnDestroy.
Contributing

Conventional commits, pre-commit hooks (lint-staged), PR template, CODEOWNERS.
Security

Report via SECURITY.md. Avoid secrets in repo.

import { AfterViewInit, Component, ElementRef, OnDestroy, ViewChild } from '@angular/core'; import * as THREE from 'three'; @Component({ selector: 'app-braid-three', templateUrl: './braid-three.component.html', }) export class BraidThreeComponent implements AfterViewInit, OnDestroy { @ViewChild('canvas', { static: true }) canvasRef!: ElementRef<HTMLCanvasElement>; private renderer!: THREE.WebGLRenderer; private scene = new THREE.Scene(); private camera = new THREE.PerspectiveCamera(75, 1, 0.1, 1000); private frameId: number | null = null; ngAfterViewInit(): void { const canvas = this.canvasRef.nativeElement; this.renderer = new THREE.WebGLRenderer({ canvas, antialias: true }); this.resize(); window.addEventListener('resize', this.resize); this.initScene(); this.animate(); } private initScene() { const ambient = new THREE.AmbientLight(0x404040); this.scene.add(ambient); const dir = new THREE.DirectionalLight(0xffffff, 1); dir.position.set(1,1,1); this.scene.add(dir); const geo = new THREE.TorusGeometry(1.5, 0.5, 16, 100); const mat = new THREE.MeshPhongMaterial({ color: 0xff0000 }); this.scene.add(new THREE.Mesh(geo, mat)); this.camera.position.z = 5; } private animate = () => { this.frameId = requestAnimationFrame(this.animate); this.renderer.render(this.scene, this.camera); }; private resize = () => { const canvas = this.canvasRef.nativeElement; const w = canvas.clientWidth || 800, h = canvas.clientHeight || 600; this.camera.aspect = w / h; this.camera.updateProjectionMatrix(); this.renderer.setSize(w, h, false); }; ngOnDestroy(): void { if (this.frameId) cancelAnimationFrame(this.frameId); window.removeEventListener('resize', this.resize); this.renderer.dispose(); } }

README.md

MillionSongMind — Angular 15
High-performance harmonic analysis and braid visualizations. Angular 15 + NgRx + Tailwind + Three.js.

Badges: CI, License, Commitizen-friendly.

Overview

Purpose: Explore tonal centers, progressions, and braid/torus topologies using deterministic state and optimized rendering.
Key Features:
Classic/Tonal/New braid visualizations (SVG + 3D).
Global key context with score/user lock.
Live metrics from corpus (presence, standalone, progressions).
SEO-first pages with canonical/meta/JSON-LD.
Reusable UI primitives themed via Tailwind semantic tokens (HSL).
Tech Stack

Angular 15, RxJS, NgRx, Tailwind, Three.js, Karma/Jasmine, Cypress, GitHub Actions.
Getting Started

Prereqs: Node 18 LTS, npm, Angular CLI 15.
Install: npm ci
Dev: npm start
Lint: npm run lint
Test: npm test
Build: npm run build
E2E (if added): npm run e2e
Architecture

Core services (SEO, data parsing, geometry).
Shared UI (headless components).
Feature modules (braid-classic, braid-tonal, braid-new2, live-metrics, braid-3d).
NgRx store slices (globalKey + router-store).
SEO

Title < 60 chars, meta description < 160 chars.
Single H1 per page. Canonical links and JSON-LD via services.
Lazy images, responsive layout.
Design System

Tailwind with semantic HSL tokens in styles.scss.
No hard-coded colors; use CSS vars.
State

NgRx with memoized selectors, OnPush, async pipes.
3D

Three.js integrated directly; careful cleanup in ngOnDestroy.
Contributing

Conventional commits, pre-commit hooks (lint-staged), PR template, CODEOWNERS.
Security

import { AfterViewInit, Component, ElementRef, OnDestroy, ViewChild } from '@angular/core'; import * as THREE from 'three'; @Component({ selector: 'app-braid-three', templateUrl: './braid-three.component.html', }) export class BraidThreeComponent implements AfterViewInit, OnDestroy { @ViewChild('canvas', { static: true }) canvasRef!: ElementRef<HTMLCanvasElement>; private renderer!: THREE.WebGLRenderer; private scene = new THREE.Scene(); private camera = new THREE.PerspectiveCamera(75, 1, 0.1, 1000); private frameId: number | null = null; ngAfterViewInit(): void { const canvas = this.canvasRef.nativeElement; this.renderer = new THREE.WebGLRenderer({ canvas, antialias: true }); this.resize(); window.addEventListener('resize', this.resize); this.initScene(); this.animate(); } private initScene() { const ambient = new THREE.AmbientLight(0x404040); this.scene.add(ambient); const dir = new THREE.DirectionalLight(0xffffff, 1); dir.position.set(1,1,1); this.scene.add(dir); const geo = new THREE.TorusGeometry(1.5, 0.5, 16, 100); const mat = new THREE.MeshPhongMaterial({ color: 0xff0000 }); this.scene.add(new THREE.Mesh(geo, mat)); this.camera.position.z = 5; } private animate = () => { this.frameId = requestAnimationFrame(this.animate); this.renderer.render(this.scene, this.camera); }; private resize = () => { const canvas = this.canvasRef.nativeElement; const w = canvas.clientWidth || 800, h = canvas.clientHeight || 600; this.camera.aspect = w / h; this.camera.updateProjectionMatrix(); this.renderer.setSize(w, h, false); }; ngOnDestroy(): void { if (this.frameId) cancelAnimationFrame(this.frameId); window.removeEventListener('resize', this.resize); this.renderer.dispose(); } }
