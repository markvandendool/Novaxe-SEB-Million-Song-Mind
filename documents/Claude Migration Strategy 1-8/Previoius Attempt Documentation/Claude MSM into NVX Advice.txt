```typescript
// packages/shared/src/bridge/IframeBridge.ts
import { Subject, Observable, filter, map } from 'rxjs';

export interface BridgeMessage<T = any> {
  type: string;
  payload: T;
  timestamp: number;
  source: 'novaxe' | 'msm';
  id: string;
}

export class IframeBridge {
  private messages$ = new Subject<BridgeMessage>();
  private pendingRequests = new Map<string, (data: any) => void>();
  private iframe: HTMLIFrameElement | null = null;
  
  constructor(private source: 'novaxe' | 'msm') {
    this.initializeListener();
  }

  private initializeListener() {
    window.addEventListener('message', (event) => {
      // Security: Validate origin
      if (!this.isValidOrigin(event.origin)) return;
      
      const message = event.data as BridgeMessage;
      
      // Handle response to pending request
      if (this.pendingRequests.has(message.id)) {
        this.pendingRequests.get(message.id)!(message.payload);
        this.pendingRequests.delete(message.id);
        return;
      }
      
      // Emit for subscribers
      this.messages$.next(message);
    });
  }

  private isValidOrigin(origin: string): boolean {
    const allowedOrigins = [
      'http://localhost:4200',
      'http://localhost:5173',
      // Add production URLs
    ];
    return allowedOrigins.includes(origin);
  }

  public on<T>(type: string): Observable<T> {
    return this.messages$.pipe(
      filter(msg => msg.type === type),
      map(msg => msg.payload as T)
    );
  }

  public async request<T, R>(type: string, payload: T): Promise<R> {
    return new Promise((resolve) => {
      const id = crypto.randomUUID();
      this.pendingRequests.set(id, resolve);
      this.send(type, payload, id);
      
      // Timeout after 5 seconds
      setTimeout(() => {
        if (this.pendingRequests.has(id)) {
          this.pendingRequests.delete(id);
          throw new Error(`Request timeout: ${type}`);
        }
      }, 5000);
    });
  }

  public send<T>(type: string, payload: T, id?: string) {
    const message: BridgeMessage<T> = {
      type,
      payload,
      timestamp: Date.now(),
      source: this.source,
      id: id || crypto.randomUUID()
    };

    if (this.iframe) {
      this.iframe.contentWindow?.postMessage(message, '*');
    } else {
      window.parent.postMessage(message, '*');
    }
  }

  public setIframe(iframe: HTMLIFrameElement) {
    this.iframe = iframe;
  }

  // Performance optimization: Batch messages
  private messageQueue: BridgeMessage[] = [];
  private batchTimer: NodeJS.Timeout | null = null;

  public sendBatched<T>(type: string, payload: T) {
    this.messageQueue.push({
      type,
      payload,
      timestamp: Date.now(),
      source: this.source,
      id: crypto.randomUUID()
    });

    if (!this.batchTimer) {
      this.batchTimer = setTimeout(() => {
        this.send('batch', this.messageQueue);
        this.messageQueue = [];
        this.batchTimer = null;
      }, 16); // Send every frame (60fps)
    }
  }
}
```

```tsx
  onMessage?: (message: BridgeMessage) => void;
}

export function useNovaxeBridge(options: UseNovaxeBridgeOptions = {}) {

  const [isConnected, setIsConnected] = useState(false);
  const [config, setConfig] = useState<any>(null);

    bridge.current = new IframeBridge('msm');

    const initSub = bridge.current.on('init').subscribe((cfg: any) => {
      setConfig(cfg);
      setIsConnected(true);
      options.onInit?.(cfg);
    });
    
    // Notify parent that MSM is ready
    bridge.current.send('ready', { timestamp: Date.now() });
    
    return () => {
      initSub.unsubscribe();
    };
  }, []);
  
  const sendToNovaxe = useCallback((type: string, payload: any) => {
    if (bridge.current && isConnected) {
      bridge.current.send(type, payload);
    }
  }, [isConnected]);
  
  const request = useCallback(async (type: string, payload: any) => {
    if (bridge.current && isConnected) {
      return bridge.current.request(type, payload);
    }
    throw new Error('Bridge not connected');
  }, [isConnected]);
  
  return {
    isConnected,
    config,
    sendToNovaxe,
    request,
    bridge: bridge.current
  };
}
```

```typescript
// packages/shared/src/monitoring/PerformanceMonitor.ts
export class PerformanceMonitor {
  private metrics: Map<string, number[]> = new Map();
  
  measure(name: string, fn: () => void | Promise<void>) {
    const start = performance.now();
    const result = fn();
    
    if (result instanceof Promise) {
      return result.finally(() => {
        this.recordMetric(name, performance.now() - start);
      });
    } else {
      this.recordMetric(name, performance.now() - start);
    }
  }
  
  private recordMetric(name: string, duration: number) {
    if (!this.metrics.has(name)) {
      this.metrics.set(name, []);
    }
    
    const values = this.metrics.get(name)!;
    values.push(duration);
    
    // Keep only last 100 measurements
    if (values.length > 100) {
      values.shift();
    }
    
    // Log slow operations
    if (duration > 100) {
      console.warn(`Slow operation detected: ${name} took ${duration.toFixed(2)}ms`);
    }
  }
  
  getStats(name: string) {
    const values = this.metrics.get(name) || [];
    if (values.length === 0) return null;
    
    const sorted = [...values].sort((a, b) => a - b);
    return {
      avg: values.reduce((a, b) => a + b, 0) / values.length,
      min: sorted[0],
      max: sorted[sorted.length - 1],
      p50: sorted[Math.floor(sorted.length * 0.5)],
      p95: sorted[Math.floor(sorted.length * 0.95)],
      p99: sorted[Math.floor(sorted.length * 0.99)]
    };
  }
}
```

```typescript
import { throttle, debounce } from 'lodash';

export class OptimizedBridge extends IframeBridge {
  private compressionWorker?: Worker;
  
  constructor(source: 'novaxe' | 'msm') {
    super(source);
    this.initializeCompression();
  }
  
  private initializeCompression() {
    if (typeof Worker !== 'undefined') {
      // Use Web Worker for compression
      this.compressionWorker = new Worker(
        new URL('./compression.worker', import.meta.url)
      );
    }
  }
  
  // Throttle high-frequency updates
  public sendThrottled = throttle(
    (type: string, payload: any) => this.send(type, payload),
    50 // Max once per 50ms
  );
  
  // Debounce user input
  public sendDebounced = debounce(
    (type: string, payload: any) => this.send(type, payload),
    300 // Wait 300ms after last call
  );
  
  // Compress large payloads
  public async sendCompressed(type: string, payload: any) {
    const serialized = JSON.stringify(payload);
    
    if (serialized.length > 10000 && this.compressionWorker) {
      // Compress in worker thread
      return new Promise((resolve) => {
        this.compressionWorker!.postMessage({ 
          action: 'compress', 
          data: serialized 
        });
        
        this.compressionWorker!.onmessage = (e) => {
          this.send(type, { 
            compressed: true, 
            data: e.data 
          });
          resolve(void 0);
        };
      });
    } else {
      this.send(type, payload);
    }
  }
}
```
