export function sendToNovaxe(payload: BridgeChordPayload) {
  try {
    const envelope = { type: 'MSM_TO_NOVAXE', payload };
    window.parent?.postMessage(envelope, NOVAXE_ORIGIN);
    window.postMessage(envelope, NOVAXE_ORIGIN); // same-window fallback

      if (!ev.data || ev.data.type !== 'NOVAXE_TO_MSM') return;
      const { payload } = ev.data;
      // handle inbound commands from Novaxe if needed
      // e.g. payload = { cmd: 'highlight', chord: 'C' }
