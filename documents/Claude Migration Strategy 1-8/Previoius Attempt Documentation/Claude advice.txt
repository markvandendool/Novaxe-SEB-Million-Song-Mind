Web MIDI API Implementation (Modern, No Dependencies)
typescript// services/web-midi.service.ts
@Injectable({ providedIn: 'root' })
export class WebMidiService {
  private midiAccess = signal<MIDIAccess | null>(null);
  private audioContext: AudioContext | null = null;
  
  async initialize(): Promise<void> {
    // Feature detection
    if (!navigator.requestMIDIAccess) {
      console.warn('Web MIDI API not supported, falling back to Web Audio');
      return this.initializeWebAudioFallback();
    }
    
    try {
      // Request MIDI access with sysex for full compatibility
      const access = await navigator.requestMIDIAccess({ sysex: true });
      this.midiAccess.set(access);
      
      // Setup audio context with autoplay handling
      this.audioContext = new (window.AudioContext || (window as any).webkitAudioContext)();
      
      // Handle autoplay policy
      document.addEventListener('click', () => {
        if (this.audioContext?.state === 'suspended') {
          this.audioContext.resume();
        }
      }, { once: true });
      
    } catch (error) {
      console.error('MIDI access denied:', error);
      return this.initializeWebAudioFallback();
    }
  }
  
  private initializeWebAudioFallback(): void {
    // Pure Web Audio fallback for browsers without MIDI
    this.audioContext = new (window.AudioContext || (window as any).webkitAudioContext)();
    console.log('Using Web Audio API fallback');
  }
  
  playNote(note: string, velocity = 127): void {
    if (this.midiAccess()) {
      // Use real MIDI
      const outputs = this.midiAccess()!.outputs;
      outputs.forEach(output => {
        output.send([0x90, this.noteToMidi(note), velocity]);
      });
    } else if (this.audioContext) {
      // Web Audio fallback
      const oscillator = this.audioContext.createOscillator();
      oscillator.frequency.value = this.noteToFrequency(note);
      oscillator.connect(this.audioContext.destination);
      oscillator.start();
      oscillator.stop(this.audioContext.currentTime + 0.5);
    }
  }
}

async function diagnoseAndFix(): Promise<void> {
