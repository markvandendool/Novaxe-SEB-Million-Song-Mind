Step 5: Create Fix Script for RxJS Imports
Create fix-angular-20-issues.sh:
bash
#!/bin/bash

# Fix RxJS imports across all TypeScript files
echo "Fixing RxJS imports..."

# Add map import to files that use it
files=(
  "src/app/services/exercises/exercise_generator/contrepoint.service.ts"
  "src/app/services/exercises/exercise_generator/melody-gen.service.ts"
  "src/app/services/exercises/exercise_generator/exo-gen.service.ts"
  "src/app/services/exercises/exercise_results/exercise-results.service.ts"
  "src/app/services/music-utils-service/music-utils.service.ts"
  "src/app/services/parsing.service.ts"
  "src/app/services/transport/transport.service.ts"
)

for file in "${files[@]}"; do
  if [ -f "$file" ]; then
    echo "Processing $file..."
    
    # Add RxJS map import if not exists
    if ! grep -q "import.*map.*from.*rxjs" "$file"; then
      sed -i.bak '1i\
import { map } from '\''rxjs/operators'\'';
' "$file"
    fi
    
    # Fix .pipe(map issues on arrays
    sed -i.bak 's/\[\([^]]*\)\]\.pipe(map(/[\1].map(/g' "$file"
    sed -i.bak 's/array\.pipe(map(/array.map(/g' "$file"
    
    # Fix method declarations
    sed -i.bak 's/public \([a-zA-Z_][a-zA-Z0-9_]*\)(/  public \1(/g' "$file"
    
    echo "Fixed $file"
  fi
done

# Add utility methods to Array prototype if needed
echo "Adding utility functions..."

# Create utility extensions file
cat > src/app/shared/array-extensions.ts << 'EOF'
declare global {
  interface Array<T> {
    get_rand_in_array(): T;
  }
  
  interface Object {
    get_rand_in_obj(): any;
  }
  
  interface Number {
    mod(n: number): number;
  }
}

Array.prototype.get_rand_in_array = function<T>(): T {
  return this[Math.floor(Math.random() * this.length)];
};

Object.prototype.get_rand_in_obj = function(): any {
  const keys = Object.keys(this);
  const randomKey = keys[Math.floor(Math.random() * keys.length)];
  return this[randomKey];
};

Number.prototype.mod = function(n: number): number {
  return ((this % n) + n) % n;
};

export {};
EOF

echo "RxJS fixes completed!"
