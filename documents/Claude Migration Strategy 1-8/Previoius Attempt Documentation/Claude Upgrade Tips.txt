    });
  }

// NEW Pattern (Angular 20 signals):
export class PianoComponent implements OnInit {
  private http = inject(HttpClient);
  notes = signal<Note[]>([]);
  notesSignal = toSignal(this.http.get<Note[]>('/api/notes'), { initialValue: [] });

    // Signals auto-update
