// libs/web-components/src/components/chord-visualizer.ts
import { LitElement, html, css } from 'lit';
import { customElement, property, state } from 'lit/decorators.js';
import { createRoot } from 'react-dom/client';
import { ChordVisualizerReact } from '@msm/components';

@customElement('chord-visualizer')
export class ChordVisualizerElement extends LitElement {
  @property({ type: Object }) chordProgression: any = null;
  @property({ type: String }) visualizationType = 'braid';
  
  private reactRoot: any;

  firstUpdated() {
    const container = this.shadowRoot?.querySelector('#react-container');
    if (container) {
      this.reactRoot = createRoot(container);
      this.renderReactComponent();
    }
  }

  updated(changedProperties: Map<string, any>) {
    if (changedProperties.has('chordProgression') || 
        changedProperties.has('visualizationType')) {
      this.renderReactComponent();
    }
  }

  private renderReactComponent() {
    if (this.reactRoot) {
      this.reactRoot.render(
        ChordVisualizerReact({
          chordProgression: this.chordProgression,
          visualizationType: this.visualizationType,
          onChordSelect: (chord: any) => {
            this.dispatchEvent(new CustomEvent('chord-selected', {
              detail: chord,
              bubbles: true
            }));
          }
        })
      );
    }
  }

  render() {
    return html`
      <div id="react-container"></div>
    `;
  }

  static styles = css`
    :host {
      display: block;
      width: 100%;
      height: 100%;
    }
    
    #react-container {
      width: 100%;
      height: 100%;
    }
  `;
}

#### 4.1 State Synchronization Bridge
Create a sophisticated state sync system:

```typescript
// libs/state-sync/src/lib/bridge.service.ts
import { Injectable } from '@angular/core';
import { BehaviorSubject, Subject } from 'rxjs';
import { webSocket, WebSocketSubject } from 'rxjs/webSocket';

interface StateUpdate {
  type: 'chord-progression' | 'file-upload' | 'visualization-mode';
  payload: any;
  timestamp: number;
  source: 'angular' | 'react';
}

@Injectable({
  providedIn: 'root'
})
export class StateBridgeService {
  private wsSubject$: WebSocketSubject<StateUpdate>;
  private stateUpdates$ = new Subject<StateUpdate>();
  
  // Shared state
  private currentChordProgression$ = new BehaviorSubject<any>(null);
  private currentVisualizationMode$ = new BehaviorSubject<string>('braid');
  private uploadedFiles$ = new BehaviorSubject<File[]>([]);
  
  constructor() {
    // Use WebSocket for real-time sync (optional, can use localStorage)
    this.wsSubject$ = webSocket('ws://localhost:8080/state-sync');
    this.setupStateSync();
  }
  
  private setupStateSync() {
    // Listen for external state updates
    this.wsSubject$.subscribe(update => {
      this.handleStateUpdate(update);
    });
    
    // Broadcast local state changes
    this.stateUpdates$.subscribe(update => {
      this.wsSubject$.next(update);
      localStorage.setItem('novaxe-state', JSON.stringify({
        chordProgression: this.currentChordProgression$.value,
        visualizationMode: this.currentVisualizationMode$.value,
        uploadedFiles: this.uploadedFiles$.value?.map(f => f.name)
      }));
    });
  }
  
  updateChordProgression(progression: any, source: 'angular' | 'react' = 'angular') {
    this.currentChordProgression$.next(progression);
    this.stateUpdates$.next({
      type: 'chord-progression',
      payload: progression,
      timestamp: Date.now(),
      source
    });
  }
  
  // Getters for reactive state
  get chordProgression$() { return this.currentChordProgression$.asObservable(); }
  get visualizationMode$() { return this.currentVisualizationMode$.asObservable(); }
  get uploadedFiles$() { return this.uploadedFiles$.asObservable(); }
}
```

  analysisMode = 'harmonic';

  hasChordProgression = false;

    private route: ActivatedRoute,
    private stateBridge: StateBridgeService

    if (msmMode) {

    this.destroy$.next();
    this.destroy$.complete();

      );
    }
  }
  
  private configureMSMMode(mode: string) {
    switch (mode) {
      case 'visualization':
        this.visualizationMode = 'braid';
        this.analysisMode = 'visual';
        break;
      case 'analysis':
        this.analysisMode = 'harmonic';
        break;

    // Enhanced file processing with music theory analysis
    files.forEach(file => {

        this.processMIDIFile(file);
      } else if (file.name.endsWith('.csv')) {
        this.processChordProgressionCSV(file);
      }
    });
  }
  
  private processMIDIFile(file: File) {
    // Convert MIDI to chord progression using existing MSM logic
    const reader = new FileReader();
    reader.onload = (e) => {
      // Process MIDI data and extract chord progression
      // This would use the existing MSM MIDI processing logic
    };
    reader.readAsArrayBuffer(file);
  }
  
  private processChordProgressionCSV(file: File) {
    // Process CSV chord data using existing MSM logic
    const reader = new FileReader();
    reader.onload = (e) => {
      // Parse CSV and create chord progression
    };
    reader.readAsText(file);

  private integrateBridgeAnalysis(analysis: any) {
