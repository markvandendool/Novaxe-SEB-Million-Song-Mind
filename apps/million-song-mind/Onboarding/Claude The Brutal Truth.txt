## 🏗️ DATA3 ARCHITECTURE SPECIFICATION

### **Core Data3 Structure** (Version 1.0)
```python
data3_record = {
    # Identity
    'song_id': str,
    'filename': str,
    'dataset_source': str,
    
    # Harmonic Core (Our Competitive Advantage)
    'harmonic_fingerprint': {
        'chroma_vector': np.array(12),
        'chroma_hash': str,                    # Unique harmonic signature
        'key_detection': {
            'key': str,                        # 'C major', 'F# minor'
            'confidence': float,               # ML confidence score
            'krumhansl_score': float,          # Rule-based validation
            'ambiguity_flag': bool             # Quality control
        },
        'voice_leading_metrics': {
            'smoothness_score': float,         # Voice leading quality
            'contrary_motion_ratio': float,    # Counterpoint analysis
            'parallel_motion_penalty': float   # Theory compliance
        }
    },
    
    # Roman Numeral Analysis (Genre-Aware)
    'roman_analysis': {
        'progressions': [                      # Time-ordered chord sequence
            {
                'timestamp': float,
                'chord_symbol': str,           # 'Cmaj7', 'Gsus4'
                'roman_numeral': str,          # 'I', 'V7', 'vi'
                'inversion': str,              # '', '6', '64'
                'function': str,               # 'tonic', 'dominant', 'subdominant'
                'context_confidence': float    # Genre-aware scoring
            }
        ],
        'genre_markers': {
            'jazz_extensions': bool,
            'classical_progressions': bool,
            'pop_cadences': bool,
            'modal_mixture': bool
        }
    },
    
    # Structural Analysis
    'structure': {
        'tempo': float,
        'time_signature': str,
        'sections': [                          # Verse, Chorus, etc.
            {
                'label': str,
                'start_time': float,
                'end_time': float,
                'key_center': str
            }
        ]
    },
    
    # Metadata
    'metadata': {
        'duration': float,
        'note_count': int,
        'chord_complexity_score': float,
        'harmonic_rhythm': float,
        'creation_timestamp': str,
        'processing_version': str
    },
    
    # Future Extensions (Data3 v2.0+)
    'audio_features': {                        # Optional, future enhancement
        'spectral_centroid': np.array,
        'mfcc': np.array,
        'rhythm_strength': float,
        'audio_chroma_alignment': float        # Symbolic vs audio correlation
    }
}
```
